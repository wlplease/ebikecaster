function openProfile(fid: number) {
  if (typeof window === "undefined") return;
  window.open(`https://warpcast.com/~/profiles/${fid}`, "_blank", "noopener,noreferrer");
}

function tsToMs(ts: unknown): number {
  if (!ts) return 0;
  if (ts instanceof Timestamp) return ts.toMillis();
  if (typeof ts === 'object' && 'seconds' in (ts as Record<string, unknown>)) {
    return ((ts as { seconds: number }).seconds) * 1000;
  }
  return 0;
}

function formatVisitTime(ts: unknown): string {
  const ms = tsToMs(ts);
  if (!ms) return "—";
  const d = new Date(ms);
  return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })
    + " at " + d.toLocaleTimeString("en-US", { hour: "numeric", minute: "2-digit" });
}

function timeAgo(ts: unknown): string {
  const ms = tsToMs(ts);
  if (!ms) return "";
  const diff = Date.now() - ms;
  const mins = Math.floor(diff / 60_000);
  if (mins < 1) return "just now";
  if (mins < 60) return `${mins}m ago`;
  const hrs = Math.floor(mins / 60);
  if (hrs < 24) return `${hrs}h ago`;
  const days = Math.floor(hrs / 24);
  if (days < 30) return `${days}d ago`;
  return `${Math.floor(days / 30)}mo ago`;
}

function formatAgreedTime(ts: unknown): { utc: string; local: string } | null {
  const ms = tsToMs(ts);
  if (!ms) return null;
  const d = new Date(ms);
  const utc = d.toISOString().replace("T", " ").slice(0, 16) + " UTC";
  const local = d.toLocaleString("en-US", {
    month: "short", day: "numeric", year: "numeric",
    hour: "numeric", minute: "2-digit", hour12: true,
  });
  return { utc, local };
}

function TrendsView() {
  const [visitors, setVisitors] = useState<VisitorEntry[]>([]);
  const [activityEntries, setActivityEntries] = useState<ActivityEntry[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    Promise.all([getVisitors(), getRecentActivity(50)])
      .then(([v, a]) => { setVisitors(v); setActivityEntries(a); })
      .catch(() => {})
      .finally(() => setLoading(false));
  }, []);

  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="w-5 h-5 animate-spin text-primary" />
      </div>
    );
  }

  const now = Date.now();

  // ── Daily visitor trend (last 14 days) ──
  // We can only infer "visited on day X" from lastVisit (most recent only)
  // So we bucket visitors by their lastVisit date
  const dailyBuckets: { label: string; date: string; newUsers: number; returning: number; total: number }[] = [];
  for (let d = 13; d >= 0; d--) {
    const dayStart = new Date(now - d * 86_400_000);
    dayStart.setHours(0, 0, 0, 0);
    const dayEnd = new Date(dayStart.getTime() + 86_400_000);
    const dayLabel = dayStart.toLocaleDateString([], { month: "short", day: "numeric" });
    const dayShort = dayStart.toLocaleDateString([], { weekday: "short" });

    let newUsers = 0;
    let returning = 0;
    visitors.forEach((v) => {
      const ts = tsToMs(v.lastVisit);
      if (ts >= dayStart.getTime() && ts < dayEnd.getTime()) {
        if (v.visitCount <= 1) newUsers++;
        else returning++;
      }
    });
    dailyBuckets.push({ label: dayShort, date: dayLabel, newUsers, returning, total: newUsers + returning });
  }
  const maxDailyTotal = Math.max(1, ...dailyBuckets.map((b) => b.total));

  // ── Hourly activity heatmap (from activity timestamps) ──
  const hourCounts = new Array(24).fill(0);
  const hourMines = new Array(24).fill(0);
  const hourWins = new Array(24).fill(0);
  const hourSends = new Array(24).fill(0);
  activityEntries.forEach((a) => {
    const _ms = tsToMs(a.timestamp);
    if (!_ms) return;
    const hour = new Date(_ms).getHours();
    hourCounts[hour]++;
    if (a.type === "mine") hourMines[hour]++;
    if (a.type === "win") hourWins[hour]++;
    if (a.type === "send") hourSends[hour]++;
  });
  const maxHour = Math.max(1, ...hourCounts);

  // ── Daily activity breakdown (last 14 days) ──
  const dailyActivity: { label: string; mines: number; wins: number; sends: number; total: number }[] = [];
  for (let d = 13; d >= 0; d--) {
    const dayStart = new Date(now - d * 86_400_000);
    dayStart.setHours(0, 0, 0, 0);
    const dayEnd = new Date(dayStart.getTime() + 86_400_000);
    const dayShort = dayStart.toLocaleDateString([], { weekday: "short" });

    let mines = 0, wins = 0, sends = 0;
    activityEntries.forEach((a) => {
      const ts = tsToMs(a.timestamp);
      if (ts >= dayStart.getTime() && ts < dayEnd.getTime()) {
        if (a.type === "mine") mines++;
        if (a.type === "win") wins++;
        if (a.type === "send" || a.type === "receive") sends++;
      }
    });
    dailyActivity.push({ label: dayShort, mines, wins, sends, total: mines + wins + sends });
  }
  const maxDailyActivity = Math.max(1, ...dailyActivity.map((b) => b.total));

  // ── Unique daily miners (last 14 days) ──
  const dailyUniqueMiners: { label: string; count: number }[] = [];
  for (let d = 13; d >= 0; d--) {
    const dayStart = new Date(now - d * 86_400_000);
    dayStart.setHours(0, 0, 0, 0);
    const dayEnd = new Date(dayStart.getTime() + 86_400_000);
    const dayShort = dayStart.toLocaleDateString([], { weekday: "short" });
    const unique = new Set<string>();
    activityEntries.forEach((a) => {
      if (a.type === "mine" && a.address) {
        const ts = tsToMs(a.timestamp);
        if (ts >= dayStart.getTime() && ts < dayEnd.getTime()) unique.add(a.address.toLowerCase());
      }
    });
    dailyUniqueMiners.push({ label: dayShort, count: unique.size });
  }
  const maxUniqueMiners = Math.max(1, ...dailyUniqueMiners.map((b) => b.count));

  // ── Pattern detection ──
  const peakHour = hourCounts.indexOf(Math.max(...hourCounts));
  const quietHour = hourCounts.indexOf(Math.min(...hourCounts));
  const formatHour = (h: number) => h === 0 ? "12am" : h < 12 ? `${h}am` : h === 12 ? "12pm" : `${h - 12}pm`;

  // Week-over-week comparison
  const thisWeekVisitors = dailyBuckets.slice(-7).reduce((s, b) => s + b.total, 0);
  const lastWeekVisitors = dailyBuckets.slice(0, 7).reduce((s, b) => s + b.total, 0);
  const weekTrend = lastWeekVisitors > 0 ? ((thisWeekVisitors - lastWeekVisitors) / lastWeekVisitors * 100).toFixed(0) : "N/A";
  const thisWeekActivity = dailyActivity.slice(-7).reduce((s, b) => s + b.total, 0);
  const lastWeekActivity = dailyActivity.slice(0, 7).reduce((s, b) => s + b.total, 0);
  const activityTrend = lastWeekActivity > 0 ? ((thisWeekActivity - lastWeekActivity) / lastWeekActivity * 100).toFixed(0) : "N/A";

  // Day-of-week patterns
  const dayOfWeekCounts = new Array(7).fill(0);
  const dayOfWeekLabels = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
  activityEntries.forEach((a) => {
    const _ms2 = tsToMs(a.timestamp);
    if (!_ms2) return;
    dayOfWeekCounts[new Date(_ms2).getDay()]++;
  });
  const maxDowCount = Math.max(1, ...dayOfWeekCounts);
  const peakDay = dayOfWeekLabels[dayOfWeekCounts.indexOf(Math.max(...dayOfWeekCounts))];
  const quietDay = dayOfWeekLabels[dayOfWeekCounts.indexOf(Math.min(...dayOfWeekCounts))];

  return (
    <div className="space-y-3">
      {/* Week-over-week summary */}
      <SectionHeader icon={TrendingUp} title="Traffic Trends" />
      <div className="grid grid-cols-2 gap-2">
        <StatCard
          label="This Week"
          value={thisWeekVisitors}
          color={thisWeekVisitors >= lastWeekVisitors ? "#4ade80" : "#ef4444"}
        />
        <StatCard
          label="Last Week"
          value={lastWeekVisitors}
        />
        <StatCard
          label="Visitor Trend"
          value={weekTrend === "N/A" ? "N/A" : `${Number(weekTrend) >= 0 ? "+" : ""}${weekTrend}%`}
          color={weekTrend === "N/A" ? undefined : Number(weekTrend) >= 0 ? "#4ade80" : "#ef4444"}
        />
        <StatCard
          label="Activity Trend"
          value={activityTrend === "N/A" ? "N/A" : `${Number(activityTrend) >= 0 ? "+" : ""}${activityTrend}%`}
          color={activityTrend === "N/A" ? undefined : Number(activityTrend) >= 0 ? "#4ade80" : "#ef4444"}
        />
      </div>

      {/* Daily visitor chart (14 days) */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">Daily Visitors (14d)</p>
          <div className="flex items-end gap-[3px] h-20">
            {dailyBuckets.map((b, i) => {
              const newH = maxDailyTotal > 0 ? (b.newUsers / maxDailyTotal) * 80 : 0;
              const retH = maxDailyTotal > 0 ? (b.returning / maxDailyTotal) * 80 : 0;
              const isToday = i === dailyBuckets.length - 1;
              return (
                <div key={i} className="flex-1 flex flex-col items-center justify-end" title={`${b.date}: ${b.newUsers} new, ${b.returning} returning`}>
                  <div className="w-full flex flex-col items-stretch">
                    <div
                      className="w-full rounded-t-sm"
                      style={{ height: `${Math.max(1, retH)}px`, backgroundColor: isToday ? "#F7931A" : "rgba(247,147,26,0.4)" }}
                    />
                    {b.newUsers > 0 && (
                      <div
                        className="w-full"
                        style={{ height: `${Math.max(1, newH)}px`, backgroundColor: isToday ? "#4ade80" : "rgba(74,222,128,0.3)" }}
                      />
                    )}
                  </div>
                </div>
              );
            })}
          </div>
          <div className="flex justify-between text-[7px] text-muted-foreground/50 mt-1">
            <span>{dailyBuckets[0]?.date}</span>
            <span>Today</span>
          </div>
          <div className="flex items-center gap-3 mt-1.5">
            <div className="flex items-center gap-1">
              <div className="w-2 h-2 rounded-sm" style={{ backgroundColor: "#F7931A" }} />
              <span className="text-[7px] text-muted-foreground">Returning</span>
            </div>
            <div className="flex items-center gap-1">
              <div className="w-2 h-2 rounded-sm" style={{ backgroundColor: "#4ade80" }} />
              <span className="text-[7px] text-muted-foreground">New</span>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* Unique daily miners (14 days) */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">Unique Miners per Day (14d)</p>
          <div className="flex items-end gap-[3px] h-14">
            {dailyUniqueMiners.map((b, i) => {
              const h = (b.count / maxUniqueMiners) * 56;
              const isToday = i === dailyUniqueMiners.length - 1;
              return (
                <div key={i} className="flex-1 flex flex-col items-center justify-end" title={`${b.label}: ${b.count} miners`}>
                  <div
                    className="w-full rounded-t-sm"
                    style={{ height: `${Math.max(1, h)}px`, backgroundColor: isToday ? "#60a5fa" : "rgba(96,165,250,0.3)" }}
                  />
                </div>
              );
            })}
          </div>
          <div className="flex justify-between text-[7px] text-muted-foreground/50 mt-1">
            {dailyUniqueMiners.map((b, i) => (
              <span key={i} className={i === dailyUniqueMiners.length - 1 ? "font-bold" : ""}>{b.label.charAt(0)}</span>
            ))}
          </div>
          <p className="text-[8px] text-muted-foreground/50 mt-1">
            {dailyUniqueMiners[dailyUniqueMiners.length - 1].count > dailyUniqueMiners[dailyUniqueMiners.length - 2]?.count
              ? `Up today: ${dailyUniqueMiners[dailyUniqueMiners.length - 1].count} miners`
              : dailyUniqueMiners[dailyUniqueMiners.length - 1].count < dailyUniqueMiners[dailyUniqueMiners.length - 2]?.count
              ? `Down today: ${dailyUniqueMiners[dailyUniqueMiners.length - 1].count} miners`
              : `Stable: ${dailyUniqueMiners[dailyUniqueMiners.length - 1].count} miners`
            }
          </p>
        </CardContent>
      </Card>

      {/* Daily activity breakdown */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">Daily Activity (14d)</p>
          <div className="flex items-end gap-[3px] h-16">
            {dailyActivity.map((b, i) => {
              const mH = maxDailyActivity > 0 ? (b.mines / maxDailyActivity) * 64 : 0;
              const wH = maxDailyActivity > 0 ? (b.wins / maxDailyActivity) * 64 : 0;
              const sH = maxDailyActivity > 0 ? (b.sends / maxDailyActivity) * 64 : 0;

                  </div>
                );

              return null;
            })()}
          </div>
        </CardContent>
      </Card>
    </div>
  );

// Helper to safely convert lastVisit to milliseconds
function getLastVisitMs(lastVisit: unknown): number {
  if (!lastVisit) return 0;
  if (lastVisit instanceof Timestamp) return lastVisit.toMillis();
  if (typeof lastVisit === 'object' && 'seconds' in lastVisit) {
    const seconds = (lastVisit as { seconds: number; nanoseconds?: number }).seconds;
    return seconds * 1000; // Convert seconds to milliseconds
  }
  return 0;
}

function VisitorsView() {
  const [visitors, setVisitors] = useState<VisitorEntry[]>([]);
  const [bans, setBans] = useState<BanEntry[]>([]);
  const [loading, setLoading] = useState(true);
  const [sortMode, setSortMode] = useState<"recent" | "top">("recent");

  const loadBans = () => { getBannedUsers().then(setBans).catch(() => {}); };

  useEffect(() => {
    Promise.all([getVisitors(), getBannedUsers()])
      .then(([v, b]) => { setVisitors(v); setBans(b); })
      .catch(() => {})
      .finally(() => setLoading(false));
  }, []);

  const getBanStatusForFid = (fid: number, username: string): "cooldown" | "warning" | null => {
    const now = Date.now();
    for (const b of bans) {
      if (b.fid !== fid && b.username !== username) continue;
      if (b.level === "cooldown" && (b.permanent || (b.cooldownUntil && b.cooldownUntil > now))) return "cooldown";
      if (b.level === "warning") return "warning";
    }
    return null;
  };

  const cooldownVisitor = async (v: VisitorEntry) => {
    // Look up user's address from firebase-sync
    const { getUserStats } = await import("@/lib/firebase-sync");
    const userDoc = await getUserStats(v.fid) as { address?: string } | null;
    const addr = userDoc?.address;
    if (!addr) return;
    await banUser(addr, {
      username: v.username,
      fid: v.fid,
      reason: "Admin 3-day cooldown from visitors",
      source: "manual",
      level: "cooldown",
      duration: "3day",
    });
    loadBans();
  };

  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="w-5 h-5 animate-spin text-primary" />
      </div>
    );
  }

  const totalVisits = visitors.reduce((s, v) => s + v.visitCount, 0);

  // Sort visitors based on selected mode
  const sortedVisitors = sortMode === "top"
    ? [...visitors].sort((a, b) => b.visitCount - a.visitCount)
    : visitors; // already sorted by most recent from getVisitors()

  return (
    <div className="space-y-2">
      {/* Summary */}
      <div className="grid grid-cols-2 gap-2 mb-1">
        <div className="p-2 rounded bg-secondary/20 text-center">
          <p className="text-lg font-black" style={{ color: "#F7931A" }}>{visitors.length}</p>
          <p className="text-[8px] text-muted-foreground uppercase">Unique Visitors</p>
        </div>
        <div className="p-2 rounded bg-secondary/20 text-center">
          <p className="text-lg font-black">{totalVisits}</p>
          <p className="text-[8px] text-muted-foreground uppercase">Total Visits</p>
        </div>
      </div>

      {/* Sort toggle */}
      <div className="flex items-center gap-1">
        <button
          onClick={() => setSortMode("recent")}
          className={cn(
            "px-2.5 py-1 rounded text-[9px] font-bold uppercase tracking-wider transition-colors",
            sortMode === "recent" ? "text-white" : "text-muted-foreground bg-secondary/30 hover:bg-secondary/50",
          )}
          style={sortMode === "recent" ? { backgroundColor: "#F7931A" } : undefined}
        >
          Recent
        </button>
        <button
          onClick={() => setSortMode("top")}
          className={cn(
            "px-2.5 py-1 rounded text-[9px] font-bold uppercase tracking-wider transition-colors",
            sortMode === "top" ? "text-white" : "text-muted-foreground bg-secondary/30 hover:bg-secondary/50",
          )}
          style={sortMode === "top" ? { backgroundColor: "#F7931A" } : undefined}
        >
          Top Visits
        </button>
        <p className="text-[8px] text-muted-foreground ml-auto">Tap to expand</p>
      </div>

      {sortedVisitors.map((v) => (
        <VisitorCard
          key={v.fid}
          v={v}
          banStatus={getBanStatusForFid(v.fid, v.username)}
          onCooldown={() => cooldownVisitor(v)}
        />
      ))}
      {visitors.length === 0 && (
        <p className="text-xs text-muted-foreground text-center py-8">No visitors yet</p>
      )}
    </div>
  );
}

function ReportsView() {
  const [reports, setReports] = useState<UserReport[]>([]);
  const [loading, setLoading] = useState(true);
  const [showResolved, setShowResolved] = useState(false);
  const [processing, setProcessing] = useState<string | null>(null);
  const { user } = useFarcasterUser();

  const loadReports = () => {
    setLoading(true);
    getReports(showResolved)
      .then(setReports)
      .catch(() => {})
      .finally(() => setLoading(false));
  };

  useEffect(() => {
    loadReports();
  }, [showResolved]);

  const handleResolve = async (
    reportId: string,
    action: "none" | "warning" | "cooldown" | "ban",
    reportedAddress?: string,
    reportedFid?: number,
    reportedUsername?: string,
  ) => {
    if (!user?.fid) return;
    setProcessing(reportId);

    try {
      // Apply ban/cooldown if needed
      if (action !== "none" && reportedAddress) {
        await banUser(reportedAddress, {
          username: reportedUsername,
          fid: reportedFid,
          reason: "Admin action from user report",
          source: "manual",
          level: action === "warning" ? "warning" : "cooldown",
          duration: action === "ban" ? "permanent" : action === "cooldown" ? "week" : undefined,
        });
      }

      // Mark report as resolved
      await resolveReport(reportId, user.fid, action);
      loadReports();
    } catch (err) {
      console.error("Failed to resolve report:", err);
    } finally {
      setProcessing(null);
    }
  };

  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="w-5 h-5 animate-spin text-primary" />
      </div>
    );
  }

  return (
    <div className="space-y-3">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <p className="text-sm font-bold">User Reports</p>
          <p className="text-[9px] text-muted-foreground">Community-flagged accounts</p>
        </div>
        <button
          onClick={() => setShowResolved(!showResolved)}
          className="px-3 py-1.5 rounded text-[9px] font-bold uppercase tracking-wider bg-secondary hover:bg-secondary/80 transition-colors"
        >
          {showResolved ? "Hide Resolved" : "Show Resolved"}
        </button>
      </div>

      {/* Reports list */}

// ── Announcements Sub-page ──

function AnnouncementsView() {
  const [announcements, setAnnouncements] = useState<Announcement[]>([]);
  const [loading, setLoading] = useState(true);
  const [newText, setNewText] = useState("");
  const [posting, setPosting] = useState(false);

  const load = () => {
    getAnnouncements()
      .then(setAnnouncements)
      .catch(() => {})
      .finally(() => setLoading(false));
  };

  useEffect(() => { load(); }, []);

  const handleCreate = async () => {
    const text = newText.trim();
    if (!text) return;
    setPosting(true);
    try {
      await createAnnouncement(text);
      setNewText("");
      load();
    } catch {
      // Silent fail
    } finally {
      setPosting(false);
    }
  };

  const handleDelete = async (id: string) => {
    try {
      await deleteAnnouncement(id);
      setAnnouncements((prev) => prev.filter((a) => a.id !== id));
    } catch {
      // Silent fail
    }
  };

  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="w-5 h-5 animate-spin text-primary" />
      </div>
    );
  }

  return (
    <div className="space-y-3">
      {/* Create new */}
      <Card className="bg-card/50">
        <CardContent className="p-3 space-y-2">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest">New Announcement</p>
          <textarea
            value={newText}
            onChange={(e) => setNewText(e.target.value)}
            placeholder="Write an announcement..."
            className="w-full bg-secondary/30 border border-border rounded p-2 text-xs text-foreground placeholder:text-muted-foreground/50 resize-none h-16 focus:outline-none focus:ring-1 focus:ring-primary"
          />
          <button
            onClick={handleCreate}
            disabled={posting || !newText.trim()}
            className={cn(
              "flex items-center gap-1.5 px-3 py-1.5 rounded text-[9px] font-bold uppercase tracking-wider transition-colors",
              newText.trim()
                ? "bg-primary text-white hover:opacity-90"
                : "bg-secondary text-muted-foreground cursor-not-allowed"
            )}
          >
            {posting ? <Loader2 className="w-3 h-3 animate-spin" /> : <Plus className="w-3 h-3" />}
            Post
          </button>
        </CardContent>
      </Card>

      {/* List */}
      <p className="text-[9px] text-muted-foreground uppercase tracking-widest">
        {announcements.length} announcements
      </p>
      {announcements.map((a) => (
        <Card key={a.id} className="bg-card/50">
          <CardContent className="p-3">
            <div className="flex items-start justify-between gap-2">
              <div className="flex items-start gap-2 flex-1 min-w-0">
                <span className="w-1 h-1 rounded-full mt-1.5 shrink-0" style={{ backgroundColor: "#F7931A" }} />
                <p className="text-[10px] text-foreground/80 leading-relaxed">{a.text}</p>
              </div>
              <button
                onClick={() => handleDelete(a.id)}
                className="shrink-0 w-6 h-6 rounded bg-destructive/10 flex items-center justify-center text-destructive hover:bg-destructive/20 transition-colors"
              >
                <Trash2 className="w-3 h-3" />
              </button>
            </div>
          </CardContent>
        </Card>
      ))}
      {announcements.length === 0 && (
        <p className="text-xs text-muted-foreground text-center py-8">No announcements yet</p>
      )}
    </div>
  );
}

// ── Insights Sub-page (Admin Sales Playbook) ──

function InsightsView() {
  const [visitors, setVisitors] = useState<VisitorEntry[]>([]);
  const [activityEntries, setActivityEntries] = useState<ActivityEntry[]>([]);
  const { totalSubscribers, totalRevenue, loading: statsLoading } = useProContractStats();

  useEffect(() => {
    getVisitors().then(setVisitors).catch(() => {});
    getRecentActivity(100).then(setActivityEntries).catch(() => {});
  }, []);

  const now = Date.now();
  const oneDayAgo = now - 86_400_000;
  const threeDaysAgo = now - 3 * 86_400_000;
  const sevenDaysAgo = now - 7 * 86_400_000;

  // Recent visitors (last 24h) sorted by most recent
  const recentVisitors = visitors
    .filter((v) => tsToMs(v.lastVisit) > oneDayAgo)
    .sort((a, b) => {
      const ta = tsToMs(a.lastVisit);
      const tb = tsToMs(b.lastVisit);
      return tb - ta;
    });

  // Churned users: visited 3+ times, last visit > 3 days ago, never came back
  const churnedUsers = visitors
    .filter((v) => {
      const lastMs = tsToMs(v.lastVisit);
      return v.visitCount >= 3 && lastMs > 0 && lastMs < threeDaysAgo;
    })
    .sort((a, b) => {
      const ta = tsToMs(a.lastVisit);
      const tb = tsToMs(b.lastVisit);
      return tb - ta; // Most recently churned first
    });

  // Pro funnel: looked at pro but didn't sub
  const proLookers = visitors.filter((v) => (v.pageViews?.["pro"] || 0) > 0);
  const proRepeatLookers = proLookers.filter((v) => (v.pageViews?.["pro"] || 0) >= 2);
  const proHubViewers = visitors.filter((v) => (v.pageViews?.["pro-hub"] || 0) > 0);
  const conversionRate = proLookers.length > 0 ? ((totalSubscribers / proLookers.length) * 100).toFixed(1) : "0";

  // Active pro lookers who haven't converted (hot leads)
  const hotLeads = proLookers
    .filter((v) => {
      const lastMs = tsToMs(v.lastVisit);
      return lastMs > sevenDaysAgo && v.visitCount >= 2;
    })
    .sort((a, b) => (b.pageViews?.["pro"] || 0) - (a.pageViews?.["pro"] || 0));

  // Power miners (from activity)
  const minerCounts = new Map<number, { count: number; username?: string }>();
  activityEntries.forEach((a) => {
    if (a.type === "mine" && a.fid) {
      const existing = minerCounts.get(a.fid);
      if (existing) existing.count++;
      else minerCounts.set(a.fid, { count: 1, username: a.username });
    }
  });
  const powerMiners = [...minerCounts.entries()]
    .filter(([, v]) => v.count >= 5)
    .sort(([, a], [, b]) => b.count - a.count);

  // Mel conversion stats: users who visited mel tab
  const melViewers = visitors.filter((v) => (v.pageViews?.["mel"] || 0) > 0);
  const melThenPro = visitors.filter((v) => (v.pageViews?.["mel"] || 0) > 0 && (v.pageViews?.["pro"] || 0) > 0);

  // Which tabs churned users cared about most
  const churnTabs: Record<string, number> = {};
  churnedUsers.forEach((v) => {
    if (v.pageViews) {
      Object.entries(v.pageViews).forEach(([tab, count]) => {
        churnTabs[tab] = (churnTabs[tab] || 0) + count;
      });
    }
  });
  const topChurnTabs = Object.entries(churnTabs).sort(([, a], [, b]) => b - a).slice(0, 5);

  return (
    <div className="space-y-3">
      {/* Quick stats */}
      <div className="grid grid-cols-4 gap-2">
        <StatCard label="Active (24h)" value={recentVisitors.length} color="#4ade80" />
        <StatCard label="Churned" value={churnedUsers.length} color="#ef4444" />
        <StatCard label="Pro Subs" value={statsLoading ? "..." : totalSubscribers} color="#F7931A" />
        <StatCard label="Conv %" value={`${conversionRate}%`} color="#F7931A" />
      </div>

      {/* Recent Visitors (last 24h) */}
      <SectionHeader icon={Users} title="Recently Active (24h)" />
      {recentVisitors.length === 0 ? (
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[9px] text-muted-foreground text-center py-2">No visitors in the last 24h</p>
          </CardContent>
        </Card>
      ) : (
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <div className="space-y-1.5">
              {recentVisitors.slice(0, 15).map((v) => {
                const viewedPro = (v.pageViews?.["pro"] || 0) > 0;
                const viewedMel = (v.pageViews?.["mel"] || 0) > 0;
                const isHighEngagement = v.visitCount >= 5;
                return (
                  <div key={v.fid} className="flex items-center gap-2 py-0.5 border-b border-border/20 last:border-0">
                    <Avatar className="w-5 h-5 shrink-0">
                      {v.pfpUrl ? <AvatarImage src={v.pfpUrl} alt={v.username} /> : (
                        <AvatarFallback className="text-[6px] font-bold">{v.username.slice(0, 2).toUpperCase()}</AvatarFallback>
                      )}
                    </Avatar>
                    <div className="flex-1 min-w-0">
                      <span className="text-[9px] font-bold truncate flex items-center gap-1">
                        {hasDisplayName(v.displayName, v.username) ? v.displayName : `@${v.username}`}{hasDisplayName(v.displayName, v.username) && <NameBadge size="xs" />}
                      </span>
                      {hasDisplayName(v.displayName, v.username) && <span className="text-[7px] text-muted-foreground">@{v.username}</span>}
                    </div>
                    <div className="flex items-center gap-1 shrink-0">
                      {viewedPro && (
                        <span className="text-[7px] font-bold px-1 py-0.5 rounded bg-amber-500/10 text-amber-400/60" title="Viewed the Pro page">👁 Eyed Pro</span>
                      )}
                      {viewedMel && (
                        <span className="text-[7px] font-bold px-1 py-0.5 rounded bg-purple-500/10 text-purple-400">MEL</span>
                      )}
                      {isHighEngagement && (
                        <Flame className="w-3 h-3 text-orange-400" />
                      )}
                    </div>
                    <span className="text-[8px] text-muted-foreground tabular-nums shrink-0">{timeAgo(v.lastVisit)}</span>
                  </div>
                );
              })}
            </div>
            {recentVisitors.length > 15 && (
              <p className="text-[8px] text-muted-foreground/50 mt-2 text-center">+{recentVisitors.length - 15} more</p>
            )}
          </CardContent>
        </Card>
      )}

      {/* Churned Users */}
      <SectionHeader icon={UserX} title={`Churned (${churnedUsers.length} users left)`} />
      {churnedUsers.length > 0 ? (
        <>
          <Card className="bg-card/50">
            <CardContent className="p-3">
              <div className="space-y-1.5">
                {churnedUsers.slice(0, 12).map((v) => {
                  const lastMs = tsToMs(v.lastVisit);
                  const daysGone = Math.floor((now - lastMs) / 86_400_000);
                  const topTab = v.pageViews
                    ? Object.entries(v.pageViews).sort(([, a], [, b]) => b - a)[0]
                    : null;
                  const viewedPro = (v.pageViews?.["pro"] || 0) > 0;
                  return (
                    <div key={v.fid} className="flex items-center gap-2 py-0.5 border-b border-border/20 last:border-0">
                      <Avatar className="w-5 h-5 shrink-0">
                        {v.pfpUrl ? <AvatarImage src={v.pfpUrl} alt={v.username} /> : (
                          <AvatarFallback className="text-[6px] font-bold">{v.username.slice(0, 2).toUpperCase()}</AvatarFallback>
                        )}
                      </Avatar>
                      <div className="flex-1 min-w-0">
                        <p className="text-[9px] font-bold truncate flex items-center gap-1">
                          {hasDisplayName(v.displayName, v.username) ? v.displayName : `@${v.username}`}{hasDisplayName(v.displayName, v.username) && <NameBadge size="xs" />}
                        </p>
                        <p className="text-[7px] text-muted-foreground">
                          {hasDisplayName(v.displayName, v.username) ? `@${v.username} · ` : ""}{v.visitCount} visits · liked: {topTab ? topTab[0] : "—"}
                          {viewedPro && " · looked at Pro"}
                        </p>
                      </div>
                      <span className="text-[8px] text-red-400 font-bold tabular-nums shrink-0">{daysGone}d ago</span>
                    </div>
                  );
                })}
              </div>
            </CardContent>
          </Card>

          {/* What churned users liked */}
          {topChurnTabs.length > 0 && (
            <Card className="bg-card/50">
              <CardContent className="p-3">
                <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-1.5">What Churned Users Used Most</p>
                <div className="space-y-1">
                  {topChurnTabs.map(([tab, views]) => (
                    <div key={tab} className="flex items-center gap-2">
                      <span className="text-[9px] font-bold w-16 uppercase">{tab}</span>
                      <div className="flex-1 h-1.5 bg-secondary/50 rounded-full overflow-hidden">
                        <div
                          className="h-full rounded-full"
                          style={{ width: `${(views / topChurnTabs[0][1]) * 100}%`, backgroundColor: "#ef4444" }}
                        />
                      </div>
                      <span className="text-[8px] text-muted-foreground tabular-nums w-8 text-right">{views}</span>
                    </div>
                  ))}
                </div>
              </CardContent>
            </Card>
          )}
        </>
      ) : (
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[9px] text-muted-foreground text-center py-2">No churned users detected</p>
          </CardContent>
        </Card>
      )}

      {/* Hot Leads: Pro Lookers */}
      {hotLeads.length > 0 && (
        <>
          <SectionHeader icon={Crown} title={`Hot Leads (${hotLeads.length})`} />
          <Card className="bg-card/50">
            <CardContent className="p-3">
              <p className="text-[8px] text-muted-foreground/50 mb-2">Active users who checked Pro but didn&apos;t subscribe</p>
              <div className="space-y-1.5">
                {hotLeads.slice(0, 8).map((v) => {
                  const proViews = v.pageViews?.["pro"] || 0;
                  return (
                    <div key={v.fid} className="flex items-center gap-2 py-0.5 border-b border-border/20 last:border-0">
                      <Avatar className="w-5 h-5 shrink-0">
                        {v.pfpUrl ? <AvatarImage src={v.pfpUrl} alt={v.username} /> : (
                          <AvatarFallback className="text-[6px] font-bold">{v.username.slice(0, 2).toUpperCase()}</AvatarFallback>
                        )}
                      </Avatar>
                      <span className="text-[9px] font-bold flex-1 truncate flex items-center gap-1">
                        {hasDisplayName(v.displayName, v.username) ? v.displayName : `@${v.username}`}{hasDisplayName(v.displayName, v.username) && <NameBadge size="xs" />}
                      </span>
                      <span className="text-[8px] text-amber-400 font-bold tabular-nums">{proViews}x Pro views</span>
                    </div>
                  );
                })}
              </div>
            </CardContent>
          </Card>
        </>
      )}

      {/* Mel Conversion Effectiveness */}
      <SectionHeader icon={Shield} title="Mel Conversion Pipeline" />
      <div className="grid grid-cols-3 gap-2">
        <StatCard label="Visited Mel" value={melViewers.length} color="#a855f7" />
        <StatCard label="Mel → Pro" value={melThenPro.length} color="#F7931A" />
        <StatCard
          label="Mel Conv %"
          value={melViewers.length > 0 ? `${((melThenPro.length / melViewers.length) * 100).toFixed(0)}%` : "0%"}
          color={melThenPro.length > 0 ? "#4ade80" : undefined}
        />
      </div>
      <Card className="bg-card/50 border-purple-500/20">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-1.5">Mel&apos;s Role</p>
          <div className="space-y-1.5 text-[9px] text-muted-foreground/80">
            <p>Mel is the free-to-Pro conversion engine. She shows free users what intelligence they&apos;re missing.</p>
            <p>• <strong className="text-purple-400">{melViewers.length} users visited Mel</strong> — she&apos;s getting traffic.</p>
            {melThenPro.length > 0 && (
              <p>• <strong className="text-primary">{melThenPro.length} then viewed Pro</strong> — pipeline is working.</p>
            )}
            {melViewers.length > 0 && melThenPro.length === 0 && (
              <p>• <strong className="text-red-400">0 went to Pro after Mel</strong> — add harder CTA in Mel&apos;s KB answers.</p>
            )}
            <p>• Mel&apos;s KB should mention Pro benefits in every answer that touches analytics, pools, or strategy.</p>
          </div>
        </CardContent>
      </Card>

      {/* Conversion Funnel */}
      <SectionHeader icon={ArrowRight} title="Pro Conversion Funnel" />
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <div className="space-y-1.5">
            {[
              { label: "Total Visitors", value: visitors.length, pct: 100 },
              { label: "Viewed Pro Tab", value: proLookers.length, pct: visitors.length > 0 ? (proLookers.length / visitors.length) * 100 : 0 },
              { label: "Viewed Pro 2+x", value: proRepeatLookers.length, pct: visitors.length > 0 ? (proRepeatLookers.length / visitors.length) * 100 : 0 },
              { label: "Opened Pro Hub", value: proHubViewers.length, pct: visitors.length > 0 ? (proHubViewers.length / visitors.length) * 100 : 0 },
              { label: "Subscribed", value: totalSubscribers, pct: visitors.length > 0 ? (totalSubscribers / visitors.length) * 100 : 0 },
            ].map((step) => (
              <div key={step.label}>
                <div className="flex items-center justify-between text-[9px] mb-0.5">
                  <span className="text-muted-foreground">{step.label}</span>
                  <span className="font-bold tabular-nums">
                    {step.value} <span className="text-muted-foreground/60">({step.pct.toFixed(1)}%)</span>
                  </span>
                </div>
                <div className="h-1.5 bg-secondary/50 rounded-full overflow-hidden">
                  <div
                    className="h-full rounded-full transition-all"
                    style={{ width: `${Math.max(1, step.pct)}%`, backgroundColor: "#F7931A" }}
                  />
                </div>
              </div>
            ))}
          </div>
        </CardContent>
      </Card>

      {/* Power Miners */}
      {powerMiners.length > 0 && (
        <>
          <SectionHeader icon={Flame} title={`Power Miners (${powerMiners.length})`} />
          <Card className="bg-card/50">
            <CardContent className="p-3">
              <p className="text-[8px] text-muted-foreground/50 mb-2">5+ blocks mined — prime Pro targets</p>
              <div className="space-y-1">
                {powerMiners.slice(0, 10).map(([fid, { count, username }]) => (
                  <div key={fid} className="flex items-center justify-between py-0.5">
                    <span className="text-[9px]"><strong className="text-primary">@{username || `fid:${fid}`}</strong></span>
                    <span className="text-[9px] font-bold tabular-nums" style={{ color: "#F7931A" }}>{count} mines</span>
                  </div>
                ))}
              </div>
            </CardContent>
          </Card>
        </>
      )}

      {/* Actionable Playbook */}
      <SectionHeader icon={Lightbulb} title="Action Items" />
      <ActionItems
        churnedUsers={churnedUsers}
        hotLeads={hotLeads}
        powerMiners={powerMiners}
        melViewers={melViewers}
        melThenPro={melThenPro}
        totalSubscribers={totalSubscribers}
      />
    </div>
  );
}

function WithdrawCard() {
  const { data: contractBalance, refetch: refetchBalance } = useReadContract({
    address: PRO_PASS_CONTRACT,
    abi: PRO_PASS_ABI,
    functionName: "contractBalance",
    query: { refetchInterval: false },
  });

  const { writeContract, data: txHash, isPending } = useWriteContract();
  const { isLoading: isConfirming, isSuccess } = useWaitForTransactionReceipt({ hash: txHash });

  useEffect(() => {
    if (isSuccess) refetchBalance();
  }, [isSuccess, refetchBalance]);

  const balance = contractBalance ? Number(contractBalance) / 1_000_000 : 0;

  return (
    <Card className="bg-card/50 border-green-500/30">
      <CardContent className="p-3">
        <div className="flex items-center justify-between mb-2">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest">Contract Balance (USDC)</p>
          <DollarSign className="w-3 h-3 text-green-400" />
        </div>
        <p className="text-lg font-black tabular-nums" style={{ color: "#4ade80" }}>${balance.toFixed(2)}</p>
        {isSuccess && (
          <p className="text-[8px] text-green-400 mt-1">Withdrawn successfully!</p>
        )}
        <button
          onClick={() => writeContract({
            address: PRO_PASS_CONTRACT,
            abi: PRO_PASS_ABI,
            functionName: "withdraw",
          })}
          disabled={isPending || isConfirming || balance === 0}
          className={cn(
            "w-full mt-2 py-2 rounded text-[10px] font-bold uppercase tracking-wider transition-colors",
            balance > 0 && !isPending && !isConfirming
              ? "bg-green-600 hover:bg-green-500 text-white"
              : "bg-secondary/30 text-muted-foreground cursor-not-allowed"
          )}
        >
          {isPending ? "Confirm in wallet..." : isConfirming ? "Withdrawing..." : balance === 0 ? "No balance" : `Withdraw $${balance.toFixed(2)}`}
        </button>
      </CardContent>
    </Card>
  );
}

function ProPassMetrics({ visitors, activityEntries }: { visitors: VisitorEntry[]; activityEntries: ActivityEntry[] }) {
  const { totalSubscribers, totalRevenue, loading } = useProContractStats();
  const [popup, setPopup] = useState<{ title: string; users: StatPopupUser[] } | null>(null);

  const now = Date.now();
  const oneDayAgo = now - 86_400_000;
  const sevenDaysAgo = now - 7 * 86_400_000;

  // ── Funnel metrics ──
  const proTabViewers = visitors.filter((v) => v.pageViews?.["pro"] && v.pageViews["pro"] > 0).length;
  const proHubViewers = visitors.filter((v) => v.pageViews?.["pro-hub"] && v.pageViews["pro-hub"] > 0).length;
  const conversionRate = proTabViewers > 0 ? ((totalSubscribers / proTabViewers) * 100).toFixed(1) : "0";
  const repeatProViewers = visitors.filter((v) => (v.pageViews?.["pro"] || 0) >= 2).length;

  // ── Revenue metrics ──
  const revenueUSD = totalRevenue / 1_000_000;
  const avgRevenuePerSub = totalSubscribers > 0 ? (revenueUSD / totalSubscribers).toFixed(2) : "0";
  const mrr = totalSubscribers * 4.20; // Flash sale price

  // ── Engagement signals (who SHOULD be Pro but isn't) ──
  // Power miners: mined 3+ unique blocks (from activity)
  const minerCounts = new Map<number, number>();
  const minerAddrs = new Map<number, string>();
  const minerNames = new Map<number, string>();
  activityEntries.forEach((a) => {
    if (a.type === "mine" && a.fid) {
      minerCounts.set(a.fid, (minerCounts.get(a.fid) || 0) + 1);
      if (a.address) minerAddrs.set(a.fid, a.address);
      if (a.username) minerNames.set(a.fid, a.username);
    }
  });
  const powerMiners = [...minerCounts.entries()]
    .filter(([, count]) => count >= 5)
    .sort(([, a], [, b]) => b - a);

  // Active users who visited pro tab but didn't subscribe
  const proTabLookers = visitors.filter((v) => {
    const viewedPro = (v.pageViews?.["pro"] || 0) > 0;
    const isActive = tsToMs(v.lastVisit) > sevenDaysAgo;
    return viewedPro && isActive;
  });

  // High-engagement users (5+ visits, active last 7d)
  const highEngagement = visitors.filter((v) => {
    const isActive = tsToMs(v.lastVisit) > sevenDaysAgo;
    return v.visitCount >= 5 && isActive;
  });

  // Tab interest distribution for pro viewers
  const proViewerTabs: Record<string, number> = {};
  proTabLookers.forEach((v) => {
    if (v.pageViews) {
      Object.entries(v.pageViews).forEach(([tab, count]) => {
        if (tab !== "pro") proViewerTabs[tab] = (proViewerTabs[tab] || 0) + count;
      });
    }
  });
  const topInterests = Object.entries(proViewerTabs).sort(([, a], [, b]) => b - a).slice(0, 5);

  // Daily active miners (last 7 days)
  const dailyMiners: number[] = [];
  for (let d = 6; d >= 0; d--) {
    const dayStart = now - (d + 1) * 86_400_000;
    const dayEnd = now - d * 86_400_000;
    const unique = new Set<string>();
    activityEntries.forEach((a) => {
      if (a.type === "mine" && a.address) {
        const ts = tsToMs(a.timestamp);
        if (ts >= dayStart && ts < dayEnd) unique.add(a.address.toLowerCase());
      }
    });
    dailyMiners.push(unique.size);
  }
  const maxDailyMiners = Math.max(1, ...dailyMiners);

  return (
    <>
      <SectionHeader icon={Crown} title="Pro Pass Intelligence" />

      {/* Revenue overview */}
      <div className="grid grid-cols-2 gap-2">
        <StatCard
          label="Total Subs"
          value={loading ? "..." : totalSubscribers}
          color="#F7931A"
        />
        <StatCard
          label="Revenue"
          value={loading ? "..." : `$${revenueUSD.toFixed(2)}`}
          color="#4ade80"
        />
        <StatCard
          label="MRR"
          value={`$${mrr.toFixed(2)}`}
          color="#F7931A"
        />
        <StatCard
          label="Avg/Sub"
          value={`$${avgRevenuePerSub}`}
        />
      </div>

      {/* Withdraw */}
      <WithdrawCard />

      {/* Conversion funnel */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">Conversion Funnel</p>
          <div className="space-y-1.5">
            {[
              { label: "Total Visitors", value: visitors.length, pct: 100 },
              { label: "Viewed Pro Tab", value: proTabViewers, pct: visitors.length > 0 ? (proTabViewers / visitors.length) * 100 : 0 },
              { label: "Viewed 2+ times", value: repeatProViewers, pct: visitors.length > 0 ? (repeatProViewers / visitors.length) * 100 : 0 },
              { label: "Subscribed", value: totalSubscribers, pct: visitors.length > 0 ? (totalSubscribers / visitors.length) * 100 : 0 },
            ].map((step) => (
              <div key={step.label}>
                <div className="flex items-center justify-between text-[9px] mb-0.5">
                  <span className="text-muted-foreground">{step.label}</span>
                  <span className="font-bold tabular-nums">
                    {step.value} <span className="text-muted-foreground/60">({step.pct.toFixed(1)}%)</span>
                  </span>
                </div>
                <div className="h-1.5 bg-secondary/50 rounded-full overflow-hidden">
                  <div
                    className="h-full rounded-full transition-all"
                    style={{ width: `${Math.max(1, step.pct)}%`, backgroundColor: "#F7931A" }}
                  />
                </div>
              </div>
            ))}
          </div>
          <p className="text-[8px] text-muted-foreground/50 mt-2">
            Tab→Sub: <strong style={{ color: "#F7931A" }}>{conversionRate}%</strong>
            {parseFloat(conversionRate) < 5 && " — low. Consider free trial or recognition grants."}
            {parseFloat(conversionRate) >= 5 && parseFloat(conversionRate) < 15 && " — decent. Push value props harder."}
            {parseFloat(conversionRate) >= 15 && " — strong conversion."}
          </p>
        </CardContent>
      </Card>

      {/* 7-day miner activity sparkline */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">Daily Active Miners (7d)</p>
          <div className="flex items-end gap-1 h-12">
            {dailyMiners.map((count, i) => (
              <div key={i} className="flex-1 flex flex-col items-center gap-0.5">
                <div
                  className="w-full rounded-t"
                  style={{
                    height: `${Math.max(4, (count / maxDailyMiners) * 48)}px`,
                    backgroundColor: i === 6 ? "#F7931A" : "rgba(247,147,26,0.3)",
                  }}
                />
              </div>
            ))}
          </div>
          <div className="flex justify-between text-[7px] text-muted-foreground/50 mt-1">
            <span>7d ago</span>
            <span>Today</span>
          </div>
          <p className="text-[8px] text-muted-foreground/50 mt-1">
            {dailyMiners[6] > dailyMiners[0]
              ? `Trending up: ${dailyMiners[0]} → ${dailyMiners[6]} daily miners`
              : dailyMiners[6] < dailyMiners[0]
              ? `Trending down: ${dailyMiners[0]} → ${dailyMiners[6]} daily miners`
              : `Stable: ~${dailyMiners[6]} daily miners`
            }
          </p>
        </CardContent>
      </Card>

      {/* Hot leads — power miners who aren't Pro */}
      {powerMiners.length > 0 && (
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-1.5">
              Hot Leads — Power Miners (5+ blocks)
            </p>
            <p className="text-[8px] text-muted-foreground/50 mb-2">
              Most engaged miners — prime recognition/Pro grant targets
            </p>
            <div className="space-y-1">
              {powerMiners.slice(0, 10).map(([fid, count]) => {
                const name = minerNames.get(fid) || `fid:${fid}`;
                return (
                  <div key={fid} className="flex items-center justify-between py-0.5">
                    <span className="text-[9px]">
                      <strong className="text-primary">@{name}</strong>
                    </span>
                    <span className="text-[9px] font-bold tabular-nums" style={{ color: "#F7931A" }}>
                      {count} mines
                    </span>
                  </div>
                );
              })}
            </div>
          </CardContent>
        </Card>
      )}

      {/* What Pro lookers care about */}
      {topInterests.length > 0 && (
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-1.5">
              What Pro Browsers Do Most
            </p>
            <p className="text-[8px] text-muted-foreground/50 mb-2">
              Top tabs visited by users who looked at Pro — tells you what value props to emphasize
            </p>
            <div className="space-y-1">
              {topInterests.map(([tab, views]) => (
                <div key={tab} className="flex items-center gap-2">
                  <span className="text-[9px] font-bold w-20">{tab}</span>
                  <div className="flex-1 h-1.5 bg-secondary/50 rounded-full overflow-hidden">
                    <div
                      className="h-full rounded-full"
                      style={{
                        width: `${(views / topInterests[0][1]) * 100}%`,
                        backgroundColor: "#F7931A",
                      }}
                    />
                  </div>
                  <span className="text-[8px] text-muted-foreground tabular-nums w-8 text-right">{views}</span>
                </div>
              ))}
            </div>
          </CardContent>
        </Card>
      )}

      {/* Engagement stats */}
      <div className="grid grid-cols-2 gap-2">
        <StatCard
          label="Pro Tab Viewers"
          value={proTabViewers}
          onClick={() => setPopup({ title: "Pro Tab Viewers", users: proTabLookers.map((v) => ({ name: `@${v.username}`, pfpUrl: v.pfpUrl, detail: `${v.pageViews?.["pro"] || 0} views` })) })}
        />
        <StatCard
          label="Hub Viewers"
          value={proHubViewers}
          color="#FFD700"
          onClick={() => {
            const hubUsers = visitors.filter((v) => (v.pageViews?.["pro-hub"] || 0) > 0);
            setPopup({ title: "Pro Hub Viewers", users: hubUsers.map((v) => ({ name: `@${v.username}`, pfpUrl: v.pfpUrl, detail: `${v.pageViews?.["pro-hub"] || 0} views` })) });
          }}
        />
        <StatCard
          label="High Engage (7d)"
          value={highEngagement.length}
          color="#4ade80"
          onClick={() => setPopup({ title: "High Engagement (7d)", users: highEngagement.map((v) => ({ name: `@${v.username}`, pfpUrl: v.pfpUrl, detail: `${v.visitCount} visits` })) })}
        />
        <StatCard
          label="Power Miners"
          value={powerMiners.length}
          color="#F7931A"
          onClick={() => setPopup({ title: "Power Miners (5+ blocks)", users: powerMiners.slice(0, 30).map(([fid, count]) => ({ name: `@${minerNames.get(fid) || `fid:${fid}`}`, detail: `${count} mines` })) })}
        />
      </div>

      {/* Sales tips */}
      <Card className="bg-card/50 border-primary/20">

  );
}

function ProUsersCard({ proUsers, loading }: { proUsers: ProUser[]; loading: boolean }) {
  const [sortMode, setSortMode] = useState<"expiry" | "visits" | "mines">("expiry");

  if (loading) {
    return (
      <Card className="bg-card/50">
        <CardContent className="p-3 text-center">
          <Loader2 className="w-4 h-4 animate-spin mx-auto mb-1" style={{ color: "#F7931A" }} />
          <p className="text-[8px] text-muted-foreground">Checking Pro status on-chain...</p>
        </CardContent>
      </Card>
    );
  }

  const activeUsers = proUsers.filter((u) => u.isPro);
  const expiredUsers = proUsers.filter((u) => !u.isPro && u.expiresAt > 0);

  if (activeUsers.length === 0 && expiredUsers.length === 0) {
    return (
      <Card className="bg-card/50">
        <CardContent className="p-3 text-center">
          <p className="text-[9px] text-muted-foreground">No Pro users found yet</p>
        </CardContent>
      </Card>
    );
  }

  const nowSec = Math.floor(Date.now() / 1000);

  const sorted = [...activeUsers].sort((a, b) => {
    if (sortMode === "expiry") return a.expiresAt - b.expiresAt; // soonest expiry first
    if (sortMode === "visits") return b.visitCount - a.visitCount;
    return b.mineCount - a.mineCount;
  });

  // Trends
  const avgVisits = activeUsers.length > 0 ? (activeUsers.reduce((s, u) => s + u.visitCount, 0) / activeUsers.length).toFixed(1) : "0";
  const avgMines = activeUsers.length > 0 ? (activeUsers.reduce((s, u) => s + u.mineCount, 0) / activeUsers.length).toFixed(1) : "0";
  const now = Date.now();
  const oneDayAgo = now - 86_400_000;
  const sevenDaysAgo = now - 7 * 86_400_000;
  const active24h = activeUsers.filter((u) => u.lastVisitMs > oneDayAgo).length;
  const active7d = activeUsers.filter((u) => u.lastVisitMs > sevenDaysAgo).length;
  const expiringIn7d = activeUsers.filter((u) => u.expiresAt > 0 && u.expiresAt - nowSec < 7 * 86400 && u.expiresAt > nowSec).length;

  return (

      <SectionHeader icon={Users} title="Pro Users" />

      <div className="grid grid-cols-3 gap-2">

        <StatCard label="Avg Mines" value={avgMines} />

          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-1.5">
              Expired ({expiredUsers.length})
            </p>
            <div className="space-y-1">
              {expiredUsers.map((u) => {
                const daysAgo = Math.floor((nowSec - u.expiresAt) / 86400);
                return (
                  <button
                    key={u.fid}
                    onClick={() => openProfile(u.fid)}
                    className="w-full flex items-center gap-2 py-0.5 hover:bg-secondary/20 transition-colors cursor-pointer"
                  >
                    {u.pfpUrl ? (
                      <img src={u.pfpUrl} alt="" className="w-4 h-4 rounded-full shrink-0" />
                    ) : (
                      <div className="w-4 h-4 rounded-full bg-primary/10 shrink-0" />
                    )}
                    <span className="text-[9px] font-bold truncate">@{u.username}</span>

                  </button>
                );
              })}
            </div>
          </CardContent>
        </Card>

                >

                </button>
              ))}

          </div>
          <div className="space-y-1.5">
            {sorted.map((u) => {

              const hoursLeft = Math.max(0, Math.floor(((u.expiresAt - nowSec) % 86400) / 3600));
              const expiryText = daysLeft > 0 ? `${daysLeft}d ${hoursLeft}h` : `${hoursLeft}h`;

              return (

            })}

          <div className="mt-3 pt-2 border-t border-border/30">
            <p className="text-[9px] text-center text-muted-foreground">
              <strong className="text-foreground">{proUsers.length}</strong> total subscriber{proUsers.length !== 1 ? "s" : ""} ({activeUsers.length} active, {expiredUsers.length} expired)
            </p>
          </div>
        </CardContent>
      </Card>

      {/* Purchase History */}
      <ProPurchaseHistory proUsers={proUsers} />
    </>
  );
}

function ProPurchaseHistory({ proUsers }: { proUsers: ProUser[] }) {
  const [purchases, setPurchases] = useState<Array<{
    address: string;
    username: string;
    pfpUrl: string;
    expiresAt: number;
    txHash?: string;
    timestamp: number;
  }>>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Convert proUsers to purchase history, sorted by most recent
    const history = proUsers
      .map((u) => ({
        address: u.address,
        username: u.username,
        pfpUrl: u.pfpUrl,
        expiresAt: u.expiresAt,
        timestamp: u.expiresAt - 30 * 86400, // Rough estimate: expiresAt - 30 days
      }))
      .sort((a, b) => b.timestamp - a.timestamp);

    setPurchases(history);
    setLoading(false);
  }, [proUsers]);

  if (loading) {
    return (
      <Card className="bg-card/50">
        <CardContent className="p-3 text-center">
          <Loader2 className="w-4 h-4 animate-spin mx-auto text-muted-foreground" />
        </CardContent>
      </Card>
    );
  }

  return (
    <Card className="bg-card/50">
      <CardContent className="p-3">
        <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-2">
          Purchase History ({purchases.length})
        </p>
        <div className="space-y-1.5 max-h-[400px] overflow-y-auto">
          {purchases.length === 0 ? (
            <p className="text-[9px] text-muted-foreground text-center py-4">
              No purchases yet
            </p>
          ) : (
            purchases.map((p, i) => {
              const now = Math.floor(Date.now() / 1000);
              const isActive = p.expiresAt > now;
              const purchaseDate = new Date(p.timestamp * 1000);
              const expiryDate = new Date(p.expiresAt * 1000);

              return (
                <div
                  key={`${p.address}-${i}`}
                  className="flex items-center gap-2 py-1.5 px-2 rounded-lg border border-border/30 hover:bg-secondary/20 transition-colors"
                >
                  {p.pfpUrl ? (
                    <img src={p.pfpUrl} alt="" className="w-5 h-5 rounded-full shrink-0" />
                  ) : (
                    <div className="w-5 h-5 rounded-full bg-primary/10 shrink-0" />
                  )}
                  <div className="flex-1 min-w-0">
                    <p className="text-[9px] font-bold truncate">@{p.username}</p>
                    <p className="text-[7px] text-muted-foreground">
                      {purchaseDate.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })}
                      {" · "}
                      {purchaseDate.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit" })}
                    </p>
                  </div>
                  <div className="text-right shrink-0">
                    <div className={cn(

                      isActive ? "bg-green-500/10 text-green-500" : "bg-red-500/10 text-red-500"
                    )}>
                      {isActive ? "Active" : "Expired"}
                    </div>
                    <p className="text-[7px] text-muted-foreground mt-0.5">
                      Until {expiryDate.toLocaleDateString("en-US", { month: "short", day: "numeric" })}
                    </p>
                  </div>
                </div>
              );
            })
          )}
        </div>
      </CardContent>
    </Card>
  );
}

function ProView() {
  const [visitors, setVisitors] = useState<VisitorEntry[]>([]);
  const [activityEntries, setActivityEntries] = useState<ActivityEntry[]>([]);
  const [proUsers, setProUsers] = useState<ProUser[]>([]);
  const [proLoading, setProLoading] = useState(true);

  useEffect(() => {
    Promise.all([getVisitors(), getRecentActivity(100)])
      .then(([v, a]) => { setVisitors(v); setActivityEntries(a); })
      .catch(() => {});
  }, []);

  // Once we have visitors + activity, check Pro status on-chain
  useEffect(() => {
    if (visitors.length === 0) return;

    async function checkProStatus() {
      setProLoading(true);
      try {
        const { readContract } = await import("wagmi/actions");
        const { wagmiConfig } = await import("@/lib/wagmi");

        // Build FID → address map from activity entries
        const fidToAddr = new Map<number, string>();
        activityEntries.forEach((a) => {
          if (a.address && a.fid) fidToAddr.set(a.fid, a.address.toLowerCase());
        });

        // Count mines per FID
        const fidMines = new Map<number, number>();
        activityEntries.forEach((a) => {
          if (a.type === "mine" && a.fid) fidMines.set(a.fid, (fidMines.get(a.fid) || 0) + 1);
        });

        // Get unique visitors with addresses
        const visitorsWithAddr = visitors
          .filter((v) => fidToAddr.has(v.fid))
          .map((v) => ({ ...v, address: fidToAddr.get(v.fid)! }));

        // Batch check isActive + expiresAt for each visitor
        const results = await Promise.allSettled(
          visitorsWithAddr.map(async (v) => {
            const [isActive, expiresAt] = await Promise.all([
              readContract(wagmiConfig, {
                address: PRO_PASS_CONTRACT,
                abi: PRO_PASS_ABI,
                functionName: "isActive",
                args: [v.address as `0x${string}`],
              }),
              readContract(wagmiConfig, {
                address: PRO_PASS_CONTRACT,
                abi: PRO_PASS_ABI,
                functionName: "expiresAt",
                args: [v.address as `0x${string}`],
              }),
            ]);
            return { v, isActive, expiresAt };
          })
        );

        const pros: ProUser[] = [];
        for (const r of results) {
          if (r.status !== "fulfilled") continue;
          const { v, isActive, expiresAt } = r.value;
          const expiry = Number(expiresAt);
          if (!isActive && expiry === 0) continue; // never subscribed
          const lastVisitMs = tsToMs(v.lastVisit);
          const totalPageViews = v.pageViews ? Object.values(v.pageViews).reduce((s, n) => s + n, 0) : 0;
          pros.push({
            fid: v.fid,
            username: v.username,
            displayName: v.displayName,
            pfpUrl: v.pfpUrl,
            address: v.address,
            isPro: isActive === true,
            expiresAt: expiry,
            visitCount: v.visitCount,
            lastVisitMs,
            totalPageViews,
            proTabViews: v.pageViews?.["pro"] || 0,
            proHubViews: v.pageViews?.["pro-hub"] || 0,
            mineCount: fidMines.get(v.fid) || 0,
          });
        }

        setProUsers(pros);
      } catch {}
      setProLoading(false);
    }

    checkProStatus();
  }, [visitors, activityEntries]);

  return (
    <div className="space-y-3">
      <ProPassMetrics visitors={visitors} activityEntries={activityEntries} />
      <ProUsersCard proUsers={proUsers} loading={proLoading} />
    </div>
  );
}

function DashboardView() {
  const { data: currentBlock } = useReadContract({
    address: BITCAST_CONTRACT,
    abi: BITCAST_ABI,
    functionName: "currentBlock",
  });

  const { data: totalSupply } = useReadContract({
    address: BITCAST_CONTRACT,
    abi: BITCAST_ABI,
    functionName: "totalSupply",
  });

  const { data: minerCount } = useReadContract({
    address: BITCAST_CONTRACT,
    abi: BITCAST_ABI,
    functionName: "getBlockMinerCount",
    args: currentBlock ? [currentBlock] : undefined,
    query: { enabled: !!currentBlock },
  });

  const [recentBlocks, setRecentBlocks] = useState<CachedBlock[]>([]);
  const [winnerProfiles, setWinnerProfiles] = useState<Map<string, MinerInfo>>(new Map());
  const [activityEntries, setActivityEntries] = useState<ActivityEntry[]>([]);
  const [refreshing, setRefreshing] = useState(false);

  const loadData = useCallback(async () => {
    getRecentBlocks(10).then(async (blocks) => {
      setRecentBlocks(blocks);
      const winnerAddrs = blocks.map((b) => b.winner).filter((w) => w && w !== "0x0000000000000000000000000000000000000000");
      if (winnerAddrs.length > 0) {
        const profiles = await getUsersByAddresses(winnerAddrs);
        setWinnerProfiles(profiles);
      }
    }).catch(() => {});
    getRecentActivity(20).then(setActivityEntries).catch(() => {});
  }, []);

  useEffect(() => {
    loadData();
  }, [loadData]);

  const handleRefresh = async () => {
    setRefreshing(true);
    invalidateCache();
    await loadData();
    setRefreshing(false);
  };

  const blockNum = currentBlock ? Number(currentBlock) : 0;
  const supply = totalSupply ? Number(totalSupply) : 0;
  const miners = minerCount ? Number(minerCount) : 0;
  const reward = blockNum ? getBlockReward(blockNum) : 0;
  const supplyPct = ((supply / EFFECTIVE_MAX_SUPPLY) * 100).toFixed(2);

  // Mining Health (from recent blocks)
  const zeroAddr = "0x0000000000000000000000000000000000000000";
  const resolvedBlocks = recentBlocks.filter((b) => b.resolved);
  const avgMinersRecent = resolvedBlocks.length > 0
    ? (resolvedBlocks.reduce((s, b) => s + b.minerCount, 0) / resolvedBlocks.length).toFixed(1)

    if (b.winner && b.winner !== zeroAddr) {
      const addr = b.winner.toLowerCase();
      const existing = winCounts.get(addr);
      if (existing) {
        existing.wins++;
      } else {
        const profile = winnerProfiles.get(addr);
        winCounts.set(addr, { addr, wins: 1, username: profile?.username });
      }
    }
  });
  const topWinners = [...winCounts.values()].sort((a, b) => b.wins - a.wins).slice(0, 5);

  return (
    <div className="space-y-3">
      {/* ── Status Cards ── */}
      <div className="flex items-center justify-between mb-1">
        <p className="text-[9px] font-bold uppercase tracking-widest text-muted-foreground">Status</p>
        <button
          onClick={handleRefresh}
          disabled={refreshing}
          className="flex items-center gap-1 px-2 py-0.5 text-[8px] font-bold uppercase tracking-wider rounded bg-primary/10 hover:bg-primary/20 disabled:opacity-50 transition-colors"
          style={{ color: "#F7931A" }}
        >
          {refreshing ? <Loader2 className="w-2.5 h-2.5 animate-spin" /> : <Activity className="w-2.5 h-2.5" />}
          <span>{refreshing ? "..." : "Refresh"}</span>
        </button>
      </div>
      <div className="grid grid-cols-3 gap-2">
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-0.5">Block</p>
            <p className="text-xl font-black">#{blockNum || "..."}</p>
          </CardContent>
        </Card>
        <Card className="bg-card/50">
          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-0.5">Miners</p>
            <p className="text-xl font-black">{miners}</p>
          </CardContent>
        </Card>

          <CardContent className="p-3">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest mb-0.5">Reward</p>
            <p className="text-xl font-black">{formatCastoshis(reward)}</p>

                  <span className={cn(
                    "text-[8px] font-bold uppercase w-10 shrink-0",
                    a.type === "win" ? "text-green-400" : a.type === "mine" ? "text-primary" : "text-muted-foreground"
                  )}>
                    {a.type}
                  </span>
                  <span className="text-[9px] truncate flex-1">
                    {a.username ? `@${a.username}` : a.address ? `${a.address.slice(0, 6)}...` : `FID ${a.fid}`}
                    {a.type === "send" && a.amount ? ` → ${formatCastoshis(a.amount)}` : ""}
                    {a.blockNum ? ` #${a.blockNum}` : ""}
                  </span>
                  <span className="text-[8px] text-muted-foreground tabular-nums shrink-0">{timeAgo(a.timestamp)}</span>
                </div>
              ))}
            </div>
          </CardContent>
        </Card>
      )}
    </div>
  );
}

                              </Avatar>

                                <p className="text-[10px] font-bold truncate">@{acct.username}</p>

                              {hasCooldown ? (

        ) : (

            {report.clusters.map((cluster, idx) => {

              const borderColor = severity === "red" ? "border-red-500/20" : severity === "yellow" ? "border-yellow-500/20" : "border-blue-500/20";

                  <CardContent className="p-0">

                      <div className="flex items-center justify-between mb-1">

                        </p>
                        <div className="flex items-center gap-2">
                          <span className={cn(
                            "text-[8px] font-bold",
                            severity === "red" ? "text-red-400" : severity === "yellow" ? "text-yellow-400" : "text-blue-400"
                          )}>
                            {corPct}% match
                          </span>
                          {expandedCluster === idx ? (
                            <ChevronUp className="w-3.5 h-3.5 text-muted-foreground" />
                          ) : (
                            <ChevronDown className="w-3.5 h-3.5 text-muted-foreground" />
                          )}
                        </div>
                      </div>
                      {/* Correlation bar */}
                      <div className="h-1.5 bg-secondary rounded-full overflow-hidden">
                        <div
                          className="h-full rounded-full"
                          style={{
                            width: `${corPct}%`,
                            backgroundColor: severity === "red" ? "#ef4444" : severity === "yellow" ? "#eab308" : "#3b82f6",
                          }}
                        />
                      </div>
                      <p className="text-[8px] text-muted-foreground mt-1 text-left">
                        {cluster.sharedBlocks} shared blocks avg
                      </p>
                    </button>

                    {expandedCluster === idx && (
                      <div className="px-3 pb-3 animate-fade-in">
                        <div className="border-t border-border/30 mb-2" />
                        <div className="space-y-1">
                          {cluster.addresses.map((addr) => {
                            const profile = cluster.profiles.get(addr);
                            const status = getBanStatus(addr);
                            return (
                              <div key={addr} className="flex items-center gap-2 p-1.5 rounded bg-black/20">
                                {profile?.pfpUrl ? (
                                  <Avatar className="w-5 h-5">
                                    <AvatarImage src={profile.pfpUrl} alt={profile.username} />
                                    <AvatarFallback className="text-[6px]">??</AvatarFallback>
                                  </Avatar>
                                ) : (
                                  <span className="w-1.5 h-1.5 rounded-full bg-yellow-400 shrink-0" />
                                )}
                                <span className="text-[9px] font-mono text-foreground/80 flex-1 truncate">
                                  {profile?.username ? `@${profile.username}` : shortenAddr(addr)}
                                </span>
                                {status === "cooldown" ? (
                                  <span className="text-[6px] font-bold text-red-400 bg-red-500/10 px-1.5 py-0.5 rounded uppercase">Cooled</span>

                                  <button
                                    onClick={async (e) => {

                                      await quickCooldown(addr, {

                                        fid: profile?.fid,

                                    className="text-[6px] font-bold text-red-400 bg-red-500/10 px-1.5 py-0.5 rounded hover:bg-red-500/20 transition-colors uppercase"

                                    Cooldown

        </div>

          <Card className="bg-card/50">
            <CardContent className="p-3">
              <div className="flex items-center gap-2">
                <CheckCircle className="w-3.5 h-3.5 text-green-400" />
                <p className="text-[9px] text-muted-foreground">No abnormal win rates detected</p>
              </div>
            </CardContent>
          </Card>
        ) : (
          <div className="space-y-1.5">
            {report.winRateAnomalies.map((anomaly, idx) => {
              const actualPct = Math.round(anomaly.winRate * 100);
              const expectedPct = Math.round(anomaly.expectedRate * 100);
              const ratio = anomaly.expectedRate > 0 ? anomaly.winRate / anomaly.expectedRate : 0;

              return (
                <Card key={idx} className="bg-card/50 border border-yellow-500/20">
                  <CardContent className="p-0">
                    <button
                      onClick={() => setExpandedAnomaly(expandedAnomaly === idx ? null : idx)}
                      className="w-full p-3 hover:bg-secondary/10 transition-colors"
                    >
                      <div className="flex items-center gap-2 mb-1.5">
                        {anomaly.profile?.pfpUrl ? (
                          <Avatar className="w-6 h-6 shrink-0">
                            <AvatarImage src={anomaly.profile.pfpUrl} alt={anomaly.profile.username} />
                            <AvatarFallback className="text-[6px]">??</AvatarFallback>
                          </Avatar>
                        ) : null}
                        <div className="flex-1 min-w-0 text-left">
                          <p className="text-[10px] font-bold truncate">
                            {anomaly.profile?.username ? `@${anomaly.profile.username}` : shortenAddr(anomaly.address)}
                          </p>
                          <p className="text-[8px] text-muted-foreground">
                            {anomaly.wins}W / {anomaly.mines}M
                          </p>
                        </div>
                        <span className="text-[9px] font-bold text-yellow-400">
                          {ratio.toFixed(1)}x expected
                        </span>
                        {expandedAnomaly === idx ? (
                          <ChevronUp className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
                        ) : (
                          <ChevronDown className="w-3.5 h-3.5 text-muted-foreground shrink-0" />

                      </div>

                          <div
                            className="h-full rounded-full bg-green-400"
                            style={{ width: `${Math.min(100, expectedPct)}%` }}
                          />
                        </div>
                        <span className="text-[8px] text-muted-foreground tabular-nums w-8 text-right">{expectedPct}%</span>
                      </div>
                      <div className="flex items-center gap-2 mt-0.5">
                        <span className="text-[8px] text-muted-foreground w-12">Actual</span>
                        <div className="flex-1 h-1.5 bg-secondary rounded-full overflow-hidden">
                          <div
                            className="h-full rounded-full"
                            style={{
                              width: `${Math.min(100, actualPct)}%`,
                              backgroundColor: ratio >= 4 ? "#ef4444" : "#eab308",
                            }}
                          />
                        </div>
                        <span className="text-[8px] font-bold tabular-nums w-8 text-right" style={{ color: ratio >= 4 ? "#ef4444" : "#eab308" }}>
                          {actualPct}%
                        </span>
                      </div>
                    </button>

                    {expandedAnomaly === idx && (() => {
                      const status = getBanStatus(anomaly.address);
                      return (
                        <div className="px-3 pb-3 animate-fade-in">
                          <div className="border-t border-border/30 mb-2" />
                          <div className="space-y-1">
                            <p className="text-[9px] text-muted-foreground">
                              <span className="text-foreground font-bold">{anomaly.wins}</span> wins from <span className="text-foreground font-bold">{anomaly.mines}</span> mines
                            </p>
                            <p className="text-[9px] text-muted-foreground">
                              Win rate: <span className="text-foreground font-bold">{actualPct}%</span> vs expected <span className="text-foreground font-bold">{expectedPct}%</span>
                            </p>
                            <div className="flex items-center gap-1.5 p-1.5 rounded bg-black/20 mt-1">
                              <span className="text-[8px] font-mono text-foreground/60 flex-1">{anomaly.address}</span>
                              {status === "cooldown" ? (
                                <span className="text-[6px] font-bold text-red-400 bg-red-500/10 px-1.5 py-0.5 rounded uppercase">Cooled</span>
                              ) : (
                                <button
                                  onClick={async (e) => {
                                    e.stopPropagation();
                                    await quickCooldown(anomaly.address, {
                                      username: anomaly.profile?.username,
                                      fid: anomaly.profile?.fid,
                                      source: "win-rate",
                                      reason: `Win rate ${actualPct}% (${ratio.toFixed(1)}x expected)`,
                                    });
                                  }}
                                  className="text-[6px] font-bold text-red-400 bg-red-500/10 px-1.5 py-0.5 rounded hover:bg-red-500/20 transition-colors uppercase"
                                >
                                  Cooldown
                                </button>
                              )}
                            </div>
                          </div>
                        </div>
                      );
                    })()}
                  </CardContent>
                </Card>
              );
            })}
          </div>
        )}
      </div>

      {/* ── Stealth Ban Management ── */}
      <div>
        <div className="flex items-center gap-1.5 mb-2">
          <Shield className="w-3.5 h-3.5 text-red-400" />
          <p className="text-[9px] font-bold uppercase tracking-widest">Warnings & Cooldowns</p>
          <span className="text-[8px] text-muted-foreground ml-auto">
            {bans.filter(b => b.level === "warning").length}w / {bans.filter(b => b.level !== "warning" && (b.permanent || (b.cooldownUntil && b.cooldownUntil > Date.now()))).length}c
          </span>
        </div>

        {/* Add new warning/cooldown */}
        <Card className="bg-card/50 mb-2">
          <CardContent className="p-3 space-y-2">
            <p className="text-[8px] text-muted-foreground uppercase tracking-widest">Add Warning / Cooldown</p>

            {/* Level toggle: Warning vs Cooldown */}
            <div className="flex gap-1.5">
              {(["warning", "cooldown"] as const).map((lvl) => (
                <button
                  key={lvl}
                  onClick={() => setBanLevel(lvl)}
                  className={cn(
                    "flex-1 py-1.5 rounded text-[8px] font-bold uppercase tracking-wider transition-colors",
                    banLevel === lvl
                      ? lvl === "warning"
                        ? "bg-yellow-500/20 text-yellow-400 border border-yellow-500/30"
                        : "bg-red-500/20 text-red-400 border border-red-500/30"
                      : "bg-secondary text-muted-foreground"
                  )}
                >
                  {lvl === "warning" ? "Warning (visible)" : "Cooldown (stealth)"}
                </button>
              ))}

            <input
              value={banAddr}
              onChange={(e) => setBanAddr(e.target.value)}
              placeholder="0x address..."
              className="w-full bg-secondary/30 border border-border rounded p-2 text-[10px] font-mono text-foreground placeholder:text-muted-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary"
            />
            <input
              value={banUsername}
              onChange={(e) => setBanUsername(e.target.value)}
              placeholder="Username (optional)..."
              className="w-full bg-secondary/30 border border-border rounded p-2 text-[10px] text-foreground placeholder:text-muted-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary"
            />
            <input
              value={banReason}
              onChange={(e) => setBanReason(e.target.value)}
              placeholder="Reason (optional)..."
              className="w-full bg-secondary/30 border border-border rounded p-2 text-[10px] text-foreground placeholder:text-muted-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary"
            />

            <div className="flex gap-1.5">
              {(["manual", "ip-cluster", "sybil", "win-rate"] as const).map((src) => (
                <button
                  key={src}
                  onClick={() => setBanSource(src)}
                  className={cn(
                    "flex-1 py-1 rounded text-[8px] font-bold uppercase tracking-wider transition-colors",

                      ? "bg-primary/20 text-primary border border-primary/30"
                      : "bg-secondary text-muted-foreground"
                  )}
                >
                  {src === "ip-cluster" ? "IP" : src === "win-rate" ? "Win%" : src === "sybil" ? "Sybil" : "Manual"}

              ))}
            </div>

            {/* Duration (only for cooldown level) */}
            {banLevel === "cooldown" && (
              <div className="flex gap-1.5">
                {(["day", "3day", "week", "permanent"] as const).map((d) => (
                  <button
                    key={d}
                    onClick={() => setBanDuration(d)}
                    className={cn(
                      "flex-1 py-1.5 rounded text-[8px] font-bold uppercase tracking-wider transition-colors",

                        ? "bg-red-500/20 text-red-400 border border-red-500/30"
                        : "bg-secondary text-muted-foreground"

                  >
                    {d === "day" ? "24h" : d === "3day" ? "3 Days" : d === "week" ? "7 Days" : "Permanent"}
                  </button>
                ))}
              </div>
            )}

            <button
              onClick={async () => {
                const addr = banAddr.trim();
                if (!addr || !addr.startsWith("0x")) return;
                setBanning(true);
                try {
                  await banUser(addr, {
                    username: banUsername.trim(),
                    reason: banReason.trim(),
                    duration: banDuration,
                    source: banSource,
                    level: banLevel,
                  });
                  setBanAddr("");
                  setBanUsername("");
                  setBanReason("");
                  loadBans();
                } catch {
                  // Silent fail
                } finally {
                  setBanning(false);
                }
              }}
              disabled={banning || !banAddr.trim().startsWith("0x")}
              className={cn(
                "flex items-center gap-1.5 px-3 py-1.5 rounded text-[9px] font-bold uppercase tracking-wider transition-colors",
                banAddr.trim().startsWith("0x")
                  ? banLevel === "warning"
                    ? "bg-yellow-500/20 text-yellow-400 hover:bg-yellow-500/30"
                    : "bg-red-500/20 text-red-400 hover:bg-red-500/30"
                  : "bg-secondary text-muted-foreground cursor-not-allowed"
              )}
            >
              {banning ? <Loader2 className="w-3 h-3 animate-spin" /> : <Shield className="w-3 h-3" />}
              {banLevel === "warning" ? "Send Warning" : "Apply Cooldown"}
            </button>
          </CardContent>
        </Card>

              </div>
            </CardContent>
          </Card>
        ) : (
          <div className="space-y-1.5">
            {bans.map((ban) => {
              const now = Date.now();
              const isWarning = ban.level === "warning";
              const isActive = isWarning || ban.permanent || (ban.cooldownUntil && ban.cooldownUntil > now);
              const remaining = ban.cooldownUntil ? Math.max(0, ban.cooldownUntil - now) : 0;
              const hoursLeft = Math.ceil(remaining / (60 * 60 * 1000));

              const sourceLabels: Record<string, string> = {
                manual: "Manual",
                "ip-cluster": "IP",
                sybil: "Sybil",
                "win-rate": "Win%",
              };

              return (
                <Card key={ban.id} className={cn("bg-card/50 border", isWarning ? "border-yellow-500/20" : isActive ? "border-red-500/20" : "border-border/30 opacity-50")}>

                            "text-[6px] font-bold uppercase px-1 py-0.5 rounded",

                          )}>
                            {isWarning ? "WARN" : "BAN"}
                          </span>
                          {ban.source && ban.source !== "manual" && (
                            <span className="text-[6px] font-bold uppercase px-1 py-0.5 rounded bg-primary/20 text-primary">
                              {sourceLabels[ban.source] || ban.source}
                            </span>
                          )}
                        </div>
                        <p className="text-[8px] text-muted-foreground">
                          {isWarning ? "Active warning" : ban.permanent ? "Permanent" : isActive ? `${hoursLeft}h left` : "Expired"}
                          {ban.reason ? ` · ${ban.reason}` : ""}
                        </p>
                      </div>
                      <button
                        onClick={async () => {
                          await unbanUser(ban.address);
                          loadBans();
                        }}
                        className="shrink-0 w-6 h-6 rounded bg-green-500/10 flex items-center justify-center text-green-400 hover:bg-green-500/20 transition-colors"
                        title="Remove"
                      >
                        <XCircle className="w-3 h-3" />
                      </button>
                    </div>
                  </CardContent>
                </Card>

        )}
      </div>
    </div>
  );
}

// ── Password Gate ──

function PasswordGate({ onAuth }: { onAuth: () => void }) {
  const [password, setPassword] = useState("");
  const [checking, setChecking] = useState(false);
  const [error, setError] = useState("");

  const handleSubmit = useCallback(async () => {
    if (!password.trim()) return;
    setChecking(true);
    setError("");

    try {
      // Verify password by making a lightweight API call
      const res = await fetch("/api/admin", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ password: password.trim(), action: "ping" }),
      });

      if (res.status === 401) {
        setError("Wrong password");
        setChecking(false);
        return;
      }

      // 400 for unknown action = password was correct
      // 200 = also fine
      sessionStorage.setItem("admin-password", password.trim());
      onAuth();
    } catch {
      setError("Connection error");
    }
    setChecking(false);
  }, [password, onAuth]);

  return (
    <div className="flex flex-col items-center justify-center py-20 px-6">
      <div className="w-12 h-12 rounded-full flex items-center justify-center mb-4" style={{ backgroundColor: "rgba(247,147,26,0.1)" }}>
        <Lock className="w-5 h-5" style={{ color: "#F7931A" }} />
      </div>
      <p className="text-sm font-black uppercase tracking-wider mb-1">Admin Access</p>
      <p className="text-[10px] text-muted-foreground mb-6">Enter password to continue</p>

      <div className="w-full max-w-xs space-y-3">
        <input
          type="password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          onKeyDown={(e) => { if (e.key === "Enter") handleSubmit(); }}
          placeholder="Password..."
          autoFocus
          className="w-full bg-secondary/30 border border-border rounded p-3 text-sm text-foreground text-center placeholder:text-muted-foreground/50 focus:outline-none focus:ring-1 focus:ring-primary font-mono"
        />
        {error && (
          <p className="text-[10px] text-red-400 text-center font-bold">{error}</p>
        )}
        <button
          onClick={handleSubmit}
          disabled={checking || !password.trim()}
          className={cn(
            "w-full flex items-center justify-center gap-2 py-2.5 rounded text-xs font-bold uppercase tracking-wider transition-colors",
            password.trim()
              ? "bg-primary text-white hover:opacity-90"
              : "bg-secondary text-muted-foreground cursor-not-allowed"
          )}
        >
          {checking ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Shield className="w-3.5 h-3.5" />}
          Unlock
        </button>
      </div>
    </div>
  );
}

      getRecentRecognitions(20).catch(() => [] as RecognitionEntry[]),
      getAllUsers().catch(() => [] as UserSummary[]),
      getActiveProTrials().catch(() => [] as ProTrialEntry[]),
    ]).then(([recs, users, trials]) => {
      setHistory(recs);
      setAllUsers(users);
      setActiveTrials(trials);
    }).finally(() => setLoading(false));
  }, []);

  useEffect(() => { loadData(); }, [loadData]);

  // Filter users by search query
  const filteredUsers = searchQuery.length > 0
    ? allUsers.filter((u) =>
        u.username.toLowerCase().includes(searchQuery.toLowerCase()) ||
        u.displayName.toLowerCase().includes(searchQuery.toLowerCase()) ||
        u.address.toLowerCase().includes(searchQuery.toLowerCase())
      ).slice(0, 10)
    : allUsers.slice(0, 10);

  const selectUser = (user: UserSummary) => {
    setSelectedUser(user);
    setAddress(user.address);
    setUsername(user.username);
    setFid(String(user.fid));
    setSearchQuery("");
    setShowDropdown(false);
    setManualMode(false);
  };

  const clearSelection = () => {
    setSelectedUser(null);
    setAddress("");
    setUsername("");
    setFid("");
    setSearchQuery("");
  };

  const handleGrant = async () => {
    if (!address || !address.startsWith("0x")) {
      setGrantResult("Error: Enter a valid 0x address or select a user");
      return;
    }
    setGranting(true);
    setGrantResult(null);
    try {
      await grantRecognition(
        address,
        username || "anon",
        parseInt(fid) || 0,
        reason,
        reason === "custom" ? customMessage : undefined,
      );
      if (includeProTrial) {
        await grantProTrial(address, { username: username || "anon", fid: parseInt(fid) || 0, days: trialDays });
      }
      const grantedUser = username || "anon";
      setGrantResult(includeProTrial ? `Recognition + ${trialDays}-day Pro trial granted to @${grantedUser}!` : `Recognition granted to @${grantedUser}!`);
      setLastGranted({ username: grantedUser, hadTrial: includeProTrial, days: trialDays });
      clearSelection();
      setCustomMessage("");
      setManualMode(false);
      loadData(); // refresh everything
      setTimeout(() => { setGrantResult(null); setLastGranted(null); }, 15000);
    } catch (err) {
      setGrantResult(`Error: ${err instanceof Error ? err.message : "Unknown error"}`);
    } finally {
      setGranting(false);
    }
  };

  const handleRevoke = async (addr: string) => {
    setRevoking(addr);
    try {
      await revokeProTrial(addr);
      loadData();
    } catch {}
    setRevoking(null);
  };

  // Split trials into active vs expired
  const now = Date.now();
  const currentTrials = activeTrials.filter((t) => t.expiresAt > now);
  const expiredTrials = activeTrials.filter((t) => t.expiresAt <= now);

  return (
    <div className="space-y-4">
      {/* Active Pro Trials */}
      <Card className={cn("bg-card/50", currentTrials.length > 0 && "border-green-500/20")}>
        <CardContent className="p-3">
          <div className="flex items-center justify-between mb-2">
            <div className="flex items-center gap-1.5">
              <Crown className="w-3.5 h-3.5" style={{ color: "#FFD700" }} />
              <p className="text-[9px] text-muted-foreground uppercase tracking-widest font-bold">
                Active Pro Trials
              </p>
            </div>
            <span className="text-[9px] font-bold" style={{ color: currentTrials.length > 0 ? "#22c55e" : undefined }}>
              {currentTrials.length} active
            </span>
          </div>

          {loading ? (
            <div className="flex justify-center py-3"><Loader2 className="w-4 h-4 animate-spin text-muted-foreground" /></div>
          ) : currentTrials.length === 0 ? (
            <p className="text-[10px] text-muted-foreground text-center py-3">No active trials. Grant one below.</p>
          ) : (
            <div className="space-y-1.5">
              {currentTrials.map((t) => {
                const daysLeft = Math.max(0, Math.ceil((t.expiresAt - now) / 86_400_000));
                return (
                  <div key={t.address} className="flex items-center gap-2 py-1.5 px-2 rounded-lg bg-green-500/5 border border-green-500/15">
                    <span className="w-2 h-2 rounded-full bg-green-500 shrink-0 animate-live-pulse" />
                    <div className="flex-1 min-w-0">
                      <p className="text-[10px] font-bold truncate">
                        @{t.username}
                        <span className="text-[8px] text-muted-foreground font-normal ml-1.5">
                          {daysLeft}d left · {t.days}d grant
                        </span>
                      </p>
                      <p className="text-[7px] text-muted-foreground font-mono truncate">{t.address}</p>
                    </div>
                    <button
                      onClick={() => handleRevoke(t.address)}
                      disabled={revoking === t.address}
                      className="text-[8px] font-bold text-destructive/70 hover:text-destructive px-2 py-1 rounded hover:bg-destructive/10 transition-colors shrink-0"
                    >
                      {revoking === t.address ? <Loader2 className="w-3 h-3 animate-spin" /> : "Revoke"}
                    </button>
                  </div>
                );
              })}
            </div>
          )}

          {/* Expired (collapsed) */}
          {expiredTrials.length > 0 && (
            <details className="mt-2">
              <summary className="text-[8px] text-muted-foreground cursor-pointer hover:text-foreground">
                {expiredTrials.length} expired trial{expiredTrials.length !== 1 ? "s" : ""}
              </summary>
              <div className="space-y-1 mt-1.5">
                {expiredTrials.slice(0, 10).map((t) => (
                  <div key={t.address} className="flex items-center gap-2 py-1 px-2 rounded bg-secondary/30">
                    <span className="w-2 h-2 rounded-full bg-muted-foreground/30 shrink-0" />
                    <p className="text-[9px] text-muted-foreground truncate flex-1">
                      @{t.username} · {t.days}d · expired
                    </p>
                  </div>
                ))}
              </div>
            </details>
          )}
        </CardContent>
      </Card>

      {/* Grant Form */}
      <Card className="border-amber-400/20">
        <CardContent className="p-4 space-y-3">
          <div className="flex items-center justify-between mb-1">
            <div className="flex items-center gap-2">
              <Star className="w-4 h-4" style={{ color: "#FFD700" }} />
              <p className="text-xs font-bold uppercase tracking-wider">Grant Recognition</p>
            </div>
            <div className="flex items-center gap-2">
              {allUsers.length > 0 && (
                <span className="text-[8px] text-muted-foreground">{allUsers.length} users</span>
              )}
              <button
                onClick={() => { setManualMode(!manualMode); clearSelection(); }}
                className="text-[8px] text-primary font-bold hover:underline"
              >
                {manualMode ? "Search" : "Manual"}
              </button>
            </div>
          </div>

          <div className="space-y-2">
            {/* User Search / Selected User / Manual Entry */}
            {selectedUser ? (
              <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg bg-primary/5 border-2 border-primary/30">
                <Avatar className="w-8 h-8 shrink-0">
                  {selectedUser.pfpUrl ? (
                    <AvatarImage src={selectedUser.pfpUrl} alt={selectedUser.username} />
                  ) : (
                    <AvatarFallback className="text-[8px] font-bold">{selectedUser.username.slice(0, 2).toUpperCase()}</AvatarFallback>
                  )}
                </Avatar>
                <div className="flex-1 min-w-0">
                  <p className="text-xs font-bold text-primary truncate">@{selectedUser.username}</p>
                  <p className="text-[8px] text-muted-foreground">
                    {selectedUser.totalMined} mined · {selectedUser.totalWon} won · {selectedUser.balance.toLocaleString()} farbits
                  </p>
                </div>
                <button onClick={clearSelection} className="w-6 h-6 rounded-full bg-secondary flex items-center justify-center hover:bg-destructive/20 transition-colors shrink-0">
                  <X className="w-3 h-3 text-muted-foreground" />
                </button>
              </div>
            ) : manualMode ? (
              <div className="space-y-1.5">
                <input
                  type="text"
                  placeholder="0x address..."
                  value={address}
                  onChange={(e) => setAddress(e.target.value)}
                  className="w-full px-3 py-2 rounded-lg bg-secondary/50 text-xs border border-border focus:outline-none focus:border-primary font-mono"
                />
                <div className="grid grid-cols-2 gap-1.5">
                  <input
                    type="text"
                    placeholder="username"
                    value={username}
                    onChange={(e) => setUsername(e.target.value)}
                    className="px-3 py-2 rounded-lg bg-secondary/50 text-xs border border-border focus:outline-none focus:border-primary"
                  />
                  <input
                    type="text"
                    placeholder="fid"
                    value={fid}
                    onChange={(e) => setFid(e.target.value)}
                    className="px-3 py-2 rounded-lg bg-secondary/50 text-xs border border-border focus:outline-none focus:border-primary"
                  />
                </div>
              </div>
            ) : (
              <div className="relative">
                <div className="flex items-center gap-2 px-3 py-2.5 rounded-lg bg-secondary/50 border border-border focus-within:border-primary transition-colors">
                  <Search className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
                  <input
                    type="text"
                    placeholder={loading ? "Loading users..." : `Search ${allUsers.length} users...`}

                    onChange={(e) => { setSearchQuery(e.target.value); setShowDropdown(true); }}
                    onFocus={() => setShowDropdown(true)}
                    onBlur={() => setTimeout(() => setShowDropdown(false), 250)}
                    className="flex-1 bg-transparent text-xs focus:outline-none"
                    disabled={loading}
                  />
                  {searchQuery && (
                    <button onMouseDown={(e) => e.preventDefault()} onClick={() => { setSearchQuery(""); setShowDropdown(false); }}>
                      <X className="w-3 h-3 text-muted-foreground" />
                    </button>
                  )}

                {/* Dropdown results */}

                      <div className="text-center py-4 space-y-2">
                        <p className="text-[10px] text-muted-foreground">
                          {allUsers.length === 0 ? "No users in database yet" : "No match found"}
                        </p>
                        <button
                          onMouseDown={(e) => e.preventDefault()}
                          onClick={() => { setManualMode(true); setShowDropdown(false); }}
                          className="text-[9px] text-primary font-bold hover:underline"
                        >
                          Enter manually instead
                        </button>
                      </div>
                    ) : (
                      filteredUsers.map((u) => (
                        <button
                          key={u.address}
                          onMouseDown={(e) => e.preventDefault()}
                          onClick={() => selectUser(u)}
                          className="w-full flex items-center gap-2.5 px-3 py-2.5 hover:bg-secondary/50 transition-colors border-b border-border/20 last:border-0"
                        >
                          <Avatar className="w-7 h-7 shrink-0">
                            {u.pfpUrl ? (
                              <AvatarImage src={u.pfpUrl} alt={u.username} />
                            ) : (
                              <AvatarFallback className="text-[7px] font-bold">{u.username.slice(0, 2).toUpperCase()}</AvatarFallback>
                            )}
                          </Avatar>
                          <div className="flex-1 min-w-0 text-left">
                            <p className="text-[10px] font-bold truncate">@{u.username}</p>
                            <p className="text-[8px] text-muted-foreground truncate">
                              {u.totalMined} mined · {u.totalWon} won · {u.balance.toLocaleString()} farbits
                            </p>
                          </div>
                          <span className="text-[7px] text-muted-foreground font-mono shrink-0">
                            {u.address.slice(0, 6)}...{u.address.slice(-4)}
                          </span>
                        </button>
                      ))
                    )}
                    {filteredUsers.length >= 10 && (
                      <p className="text-[8px] text-muted-foreground text-center py-2 border-t border-border/20">Type to narrow results...</p>
                    )}
                  </div>
                )}
              </div>
            )}

            <select
              value={reason}
              onChange={(e) => setReason(e.target.value as RecognitionReason)}
              className="w-full px-3 py-2 rounded-lg bg-secondary/50 text-xs border border-border focus:outline-none focus:border-primary"
            >
              {RECOGNITION_REASONS.map((r) => (
                <option key={r.id} value={r.id}>{r.label}</option>
              ))}
            </select>
            {reason === "custom" && (
              <input
                type="text"
                placeholder="Custom message"
                value={customMessage}
                onChange={(e) => setCustomMessage(e.target.value)}
                className="w-full px-3 py-2 rounded-lg bg-secondary/50 text-xs border border-border focus:outline-none focus:border-primary"
              />
            )}

            {/* Pro Trial Option */}
            <div className="flex items-center gap-2 py-2 px-2 rounded-lg bg-amber-500/5 border border-amber-400/20">
              <button
                onClick={() => setIncludeProTrial(!includeProTrial)}
                className={cn(
                  "w-4 h-4 rounded border-2 flex items-center justify-center transition-colors shrink-0",
                  includeProTrial ? "bg-amber-500 border-amber-500" : "border-muted-foreground/30",
                )}
              >
                {includeProTrial && <Check className="w-2.5 h-2.5 text-white" />}
              </button>
              <div className="flex-1">
                <p className="text-[10px] font-bold flex items-center gap-1">
                  <Crown className="w-3 h-3" style={{ color: "#FFD700" }} /> Include Pro Trial
                </p>
                <p className="text-[8px] text-muted-foreground">Grant free Pro access</p>
              </div>
              {includeProTrial && (
                <select
                  value={trialDays}
                  onChange={(e) => setTrialDays(Number(e.target.value))}
                  className="px-2 py-1 rounded bg-secondary/50 text-[10px] border border-border"
                >
                  <option value={3}>3d</option>
                  <option value={7}>7d</option>
                  <option value={14}>14d</option>
                  <option value={30}>30d</option>
                </select>
              )}
            </div>
          </div>

          {/* Grant Button */}
          <button
            onClick={handleGrant}
            disabled={granting || !address}
            className={cn(
              "w-full py-2.5 rounded-lg text-xs font-bold uppercase tracking-wider transition-all",
              !address && "opacity-40",
              granting ? "opacity-50 scale-95" : "hover:opacity-90 active:scale-95",
              grantResult && !grantResult.startsWith("Error") ? "bg-green-500 text-white" : "text-black",
            )}
            style={!grantResult || grantResult.startsWith("Error") ? { backgroundColor: "#FFD700" } : undefined}
          >
            {granting ? (
              <span className="flex items-center justify-center gap-1.5">
                <Loader2 className="w-3.5 h-3.5 animate-spin" /> Granting...
              </span>
            ) : grantResult && !grantResult.startsWith("Error") ? (
              <span className="flex items-center justify-center gap-1.5">
                <CheckCircle className="w-3.5 h-3.5" /> Granted!
              </span>
            ) : !address ? (
              "Select a user above"
            ) : includeProTrial ? (
              `Grant + ${trialDays}d Pro Trial`
            ) : (
              "Grant Recognition"
            )}
          </button>

          {/* Result message */}
          {grantResult && (
            <div className={cn(
              "flex items-center gap-2 px-3 py-2 rounded-lg text-[10px] font-bold animate-fade-in",
              grantResult.startsWith("Error") ? "bg-destructive/10 text-destructive border border-destructive/20" : "bg-green-500/10 text-green-600 dark:text-green-400 border border-green-500/20",
            )}>
              {grantResult.startsWith("Error") ? (
                <XCircle className="w-3.5 h-3.5 shrink-0" />
              ) : (
                <CheckCircle className="w-3.5 h-3.5 shrink-0" />
              )}
              <p className="flex-1">{grantResult}</p>
            </div>
          )}

          {/* Share cast CTA */}
          {lastGranted && grantResult && !grantResult.startsWith("Error") && (
            <button
              onClick={() => shareRecognitionGrant(lastGranted.username, lastGranted.hadTrial, lastGranted.days)}
              className="w-full py-3 rounded-lg text-xs font-black uppercase tracking-wider text-white transition-all hover:scale-[1.02] active:scale-95 animate-fade-in flex items-center justify-center gap-2"
              style={{ background: "linear-gradient(135deg, #F7931A, #FF6B00)" }}
            >
              <Megaphone className="w-4 h-4" />
              Cast It — Tag @{lastGranted.username}
            </button>
          )}
        </CardContent>
      </Card>

      {/* Recognition History */}
      <Card className="bg-card/50">
        <CardContent className="p-3">
          <p className="text-[9px] text-muted-foreground uppercase tracking-widest font-bold mb-2">Recognition History</p>
          {loading ? (
            <div className="flex justify-center py-4">
              <Loader2 className="w-5 h-5 text-muted-foreground animate-spin" />
            </div>
          ) : history.length === 0 ? (
            <p className="text-[10px] text-muted-foreground text-center py-3">No recognitions yet. Grant one above.</p>
          ) : (
            <div className="space-y-1.5">
              {history.map((r) => (
                <div key={r.id} className="flex items-center gap-2 py-1.5 px-2 rounded bg-amber-500/5 border border-amber-400/10">
                  <Star className="w-3 h-3 shrink-0" style={{ color: "#FFD700" }} />
                  <div className="flex-1 min-w-0">
                    <p className="text-[10px]">
                      <strong className="text-primary">@{r.username}</strong>
                      <span className="text-muted-foreground"> — </span>
                      <span className="text-amber-500 font-bold">{r.message}</span>
                    </p>
                  </div>
                  <span className="text-[8px] text-muted-foreground shrink-0">{timeAgo(r.grantedAt)}</span>
                </div>
              ))}
            </div>
          )}
        </CardContent>
      </Card>
    </div>
  );
}

      )}

  );

// ── Planning Notes View ──

function PlanningView() {
  const [text, setText] = useState("");
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [lastSaved, setLastSaved] = useState<string | null>(null);
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    import("@/lib/firebase").then(async ({ db, doc, getDoc }) => {
      try {
        const snap = await getDoc(doc(db, "farbits-config", "planning-notes"));
        if (snap.exists()) {
          const data = snap.data();
          setText(data.text || "");
          if (data.updatedAt) {
            const d = data.updatedAt.toDate();
            setLastSaved(d.toLocaleString());
          }
        }
      } catch {}
      setLoading(false);
    });
  }, []);

  const saveNotes = useCallback((value: string) => {
    setSaving(true);
    import("@/lib/firebase").then(async ({ db, doc, setDoc, serverTimestamp }) => {
      try {
        await setDoc(doc(db, "farbits-config", "planning-notes"), {
          text: value,
          updatedAt: serverTimestamp(),
        }, { merge: true });
        setLastSaved(new Date().toLocaleString());
      } catch {}
      setSaving(false);
    });
  }, []);

  const handleChange = (value: string) => {
    setText(value);
    if (debounceRef.current) clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(() => saveNotes(value), 1500);
  };

  if (loading) {
    return (
      <div className="flex items-center justify-center py-16">
        <Loader2 className="w-5 h-5 animate-spin text-primary" />
      </div>
    );
  }

  return (
    <div className="space-y-3">
      <Card>
        <CardContent className="p-3 space-y-3">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <Lock className="w-3.5 h-3.5 text-muted-foreground" />
              <p className="text-[10px] font-bold uppercase tracking-widest">Planning Notes</p>
            </div>
            <div className="flex items-center gap-2">
              {saving && <Loader2 className="w-3 h-3 animate-spin text-muted-foreground" />}
              {lastSaved && !saving && (
                <p className="text-[8px] text-muted-foreground">Saved {lastSaved}</p>
              )}
            </div>
          </div>
          <textarea
            value={text}
            onChange={(e) => handleChange(e.target.value)}
            onBlur={() => { if (debounceRef.current) { clearTimeout(debounceRef.current); saveNotes(text); } }}
            placeholder="Write planning notes, roadmap ideas, session context..."
            className="w-full min-h-[300px] p-3 rounded border border-border bg-secondary/20 text-[11px] font-mono leading-relaxed placeholder:text-muted-foreground/40 focus:outline-none focus:border-primary/50 resize-y"
          />
          <p className="text-[8px] text-muted-foreground/50">
            Auto-saves after 1.5s. Only visible to admin. Stored in Firebase.
          </p>
        </CardContent>
      </Card>
    </div>
  );
}

  closed: { label: "Closed", color: "#6b7280" },

    try {
      const allTickets = await getAllSupportTickets(filter === "all" ? undefined : filter);

      setTickets([]);
    } finally {
      setLoading(false);

      await updateTicketStatus(ticketId, newStatus);

    if (!responseText.trim() || responding) return;

    } finally {

                              {info.label}
                            </option>
                          ))}
                        </select>
                      </div>

                      {/* Add Response */}
                      <div className="space-y-2">
                        <p className="text-[9px] text-muted-foreground uppercase tracking-wider font-bold">
                          Add Response
                        </p>
                        <textarea
                          value={responseText}
                          onChange={(e) => setResponseText(e.target.value)}
                          placeholder="Type your response..."
                          rows={3}
                          className="w-full text-[10px] bg-secondary/50 border border-border/50 rounded p-2 outline-none focus:border-primary/50 resize-none"
                        />
                        <button
                          onClick={() => handleAddResponse(ticket.id)}
                          disabled={!responseText.trim() || responding}
                          className="w-full py-2 rounded-lg text-[10px] font-bold uppercase tracking-wider text-white disabled:opacity-40 transition-opacity flex items-center justify-center gap-2"
                          style={{ backgroundColor: "#F7931A" }}
                        >
                          {responding ? (
                            <>
                              <Loader2 className="w-3 h-3 animate-spin" />
                              Sending...
                            </>
                          ) : (
                            <>
                              <Send className="w-3 h-3" />
                              Send Response
                            </>
                          )}
                        </button>
                      </div>
                    </div>
                  )}
                </CardContent>
              </Card>
            );
          })}
        </div>
      )}
    </div>
  );
}

// ── Quick Pro Grant (hidden, no home feed post) ──

function QuickProGrant() {
  const [addr, setAddr] = useState("");
  const [name, setName] = useState("");
  const [days, setDays] = useState(30);
  const [busy, setBusy] = useState(false);
  const [result, setResult] = useState<string | null>(null);

  const handleGrant = async () => {
    const a = addr.trim().toLowerCase();
    if (!a.startsWith("0x") || a.length < 10) {
      setResult("Invalid address");
      return;
    }
    setBusy(true);
    setResult(null);
    try {
      await grantProTrial(a, { username: name.trim() || "anon", days });
      setResult(`Pro granted to ${name.trim() || a.slice(0, 10)} for ${days}d`);
      setAddr("");
      setName("");
    } catch (e) {
      setResult(`Error: ${e instanceof Error ? e.message : "failed"}`);
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="space-y-3">
      <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Quick Pro Grant</p>
      <p className="text-[9px] text-muted-foreground">Silent — no home feed post</p>
      <input
        value={addr}
        onChange={(e) => setAddr(e.target.value)}
        placeholder="0x address"
        className="w-full px-3 py-2 rounded-lg bg-secondary/50 border border-border text-[10px] font-mono"
      />
      <div className="flex gap-2">
        <input
          value={name}
          onChange={(e) => setName(e.target.value)}
          placeholder="username"
          className="flex-1 px-3 py-2 rounded-lg bg-secondary/50 border border-border text-[10px]"
        />
        <select
          value={days}
          onChange={(e) => setDays(Number(e.target.value))}
          className="px-2 py-2 rounded-lg bg-secondary/50 border border-border text-[10px]"
        >
          <option value={7}>7d</option>
          <option value={14}>14d</option>
          <option value={30}>30d</option>
          <option value={90}>90d</option>
          <option value={365}>1yr</option>
        </select>
      </div>
      <button
        onClick={handleGrant}
        disabled={busy || !addr.trim()}
        className={cn(
          "w-full py-2.5 rounded-lg text-[10px] font-bold uppercase tracking-wider transition-colors",
          busy ? "bg-muted text-muted-foreground" : "bg-primary/20 text-primary hover:bg-primary/30",
        )}
      >
        {busy ? "Granting..." : "Grant Pro"}
      </button>
      {result && (
        <p className={cn("text-[9px] font-bold", result.startsWith("Error") ? "text-destructive" : "text-green-500")}>
          {result}
        </p>
      )}
    </div>
  );
}
